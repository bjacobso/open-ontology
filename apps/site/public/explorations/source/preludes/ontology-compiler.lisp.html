1; ontology-compiler.lisp
2; -----------------------------------------------------------------------------
3; Elaboration hooks for ontology domain forms.
4;
5; This file contains meta-fn implementations that compile ontology forms
6; into canonical IR. Each hook corresponds to a (define-form ...) declaration
7; in ontology.lisp.
8;
9; Depends on: compiler.lisp (define-form, meta-fn, meta/* builtins)
10;
11; How to read this file:
12; - Each `meta-fn` name is referenced from ontology.lisp via `:*-fn` slots.
13; - `(:kind bindings)` returns names and types introduced by a form.
14; - `(:kind validate)` returns diagnostics without constructing IR.
15; - `(:kind result-type)` computes the type of the form expression.
16; - `(:kind construct)` builds the canonical IR payload consumed by TypeScript.
17; - Helper calls with `meta/*`, `type/*`, `diag/*`, and `construct/*` are
18;   compiler-hosted builtins, not user ontology functions.
19;
20; Debugging path:
21; 1. Inspect the form declaration in ontology.lisp.
22; 2. Find the hook named by `:bindings-fn`, `:validate-fn`, `:result-type-fn`,
23;    or `:construct-fn`.
24; 3. Follow the hook body here to see how slots and identifiers lower to IR.
25;
26; Status: executable (meta-fn bodies run at compile time via kernel evaluator)
27; -----------------------------------------------------------------------------

79; Entity bindings publish schema field types into the local semantic environment
80; so later forms can project source-local declaration fields through generic
81; declaration reflection instead of engine-side special cases.
82(meta-fn entity/bindings
83  (:kind bindings)
84  (:input FormMetaInput)
85  (:output BindingMap)
86  (:doc "Publishes source-local entity field bindings for later descriptor hooks.")
87  (:body
88    (let [name (meta/declaration-name input)
89          fields (meta/declaration-fields input)]
90      (if (nil? name)
91        (bindings/empty)
92        (bindings/from-fields (str "$entity:" name ":") fields)))))

94; Query bindings introduce the query declaration name, the implicit `it` row
95; binding, and query-body field bindings. The field bindings keep predicates
96; like `employee/active` descriptor-authored while row projection matures.
97(meta-fn query/bindings
98  (:kind bindings)
99  (:input FormMetaInput)
100  (:output BindingMap)
101  (:doc "Computes HM bindings for a query form.")
102  (:body
103    (let [decl-name (meta/declaration-name input)
104          from-name (meta/slot-symbol input :from)
105          from-decl (meta/lookup-declaration input from-name)
106          row-type (meta/declaration-type from-decl)]
107      (bindings/merge
108        (bindings/when decl-name
109          (bindings/from-declaration decl-name (type/constant "QueryDef")))
110        (if row-type
111          (bindings/merge
112            (bindings/of ["it" row-type])
113            (bindings/from-fields "" (meta/declaration-fields from-decl)))
114          (bindings/empty))))))

116; Query result type is either a list of full entity rows or a list of projected
117; row shapes when `:select` is present.
118(meta-fn query/result-type
119  (:kind result-type)
120  (:input FormMetaInput)
121  (:output Type)
122  (:doc "Computes the result type for a query form from :from and :select.")
123  (:body
124    (let [from-name (meta/slot-symbol input :from)
125          from-decl (meta/lookup-declaration input from-name)
126          row-type (meta/declaration-type from-decl)
127          select-fields (meta/slot-string-list input :select)]
128      (if (nil? row-type)
129        (type/unknown)
130        (if (empty? select-fields)
131          (type/list row-type)
132          (type/list (meta/project-type from-decl select-fields)))))))

134; Query validation keeps the form-level descriptor small while still allowing
135; semantic checks that need type information.
136(meta-fn query/validate
137  (:kind validate)
138  (:input FormMetaInput)
139  (:output DiagnosticList)
140  (:doc "Runs semantic validation for query instances.")
141  (:body
142    (let [where-expr (meta/slot-expr input :where)]
143      (diag/concat
144        (if (nil? where-expr)
145          []
146          (if (meta/expr-assignable-to? input where-expr "Bool")
147            []
148            [(diag/error
149               :slot :where
150               :message "query :where must evaluate to Bool")]))
151        (meta/validate-query-select-fields input)))))

153; Query inference returns the declaration type and delegates expression checking
154; to the generic meta bridge. This keeps `:where` off descriptor typed slots.
155(meta-fn query/infer
156  (:kind infer)
157  (:input FormMetaInput)
158  (:output Type)
159  (:doc "Infers QueryDef while checking :where through meta/check-expr.")
160  (:body
161    (let [where-expr (meta/slot-expr input :where)]
162      (if (nil? where-expr)
163        (type/constant "QueryDef")
164        (do
165          (meta/check-expr input where-expr (type/constant "Bool"))
166          (type/constant "QueryDef"))))))

168; Record checking validates each field assignment against the referenced entity
169; schema through the generic declaration reflection and meta/check-expr bridge.
170(meta-fn record/check
171  (:kind check)
172  (:input FormMetaInput)
173  (:output Type)
174  (:doc "Checks record field assignments against the referenced entity schema.")
175  (:body
176    (let [entity-name (meta/identifier input :entity)
177          entity-decl (meta/lookup-declaration input entity-name)
178          field-forms (meta/child-forms input :field)]
179      (do
180        (reduce
181          (fn [ok field]
182            (let [field-name (meta/identifier field :name)
183                  field-decl (meta/declaration-field entity-decl field-name)
184                  field-type (meta/declaration-type field-decl)
185                  value-expr (meta/slot-expr field :type)]
186              (if (nil? field-type)
187                ok
188                (if (nil? value-expr)
189                  ok
190                  (do
191                    (meta/check-expr input value-expr field-type)
192                    ok)))))
193          true
194          field-forms)
195        true))))

197; Query construction is where surface syntax becomes canonical compiler IR.
198; The output object keys must match the generated OntologyIR contract.
199(meta-fn query/construct
200  (:kind construct)
201  (:input FormMetaInput)
202  (:output QueryConstruct)
203  (:doc "Constructs the query IR payload for elaboration.")
204  (:body
205    (let [from-name (meta/slot-string input :from)
206          from-decl (meta/lookup-declaration input from-name)
207          select-fields (meta/query-select-fields input)
208          where-field (meta/slot-string input :where)
209          where-type (if (nil? where-field)
210            nil
211            (meta/declaration-type (meta/declaration-field from-decl where-field)))
212          select-types (reduce
213            (fn [acc field-name]
214              (assoc acc field-name
215                (meta/declaration-type (meta/declaration-field from-decl field-name))))
216            {}
217            select-fields)]
218      (construct/object
219        :kind "Query"
220        :name (let [n (meta/declaration-name input)] (if n n "anonymous-query"))
221        :$summary (construct/summary
222          :kind "Query"
223          :name (let [n (meta/declaration-name input)] (if n n "anonymous-query"))
224          :resultType "List")
225        :from from-name
226        :from-ref (meta/slot-ref input :from "Entity")
227        :select select-fields
228        :where (meta/slot-runtime-expr input :where)
229        :typeAnnotations (construct/object
230          :where where-type
231          :select select-types)
232        :loc (meta/loc input)))))

365(meta-fn process/validate
366  (:kind validate)
367  (:input FormMetaInput)
368  (:output DiagnosticList)
369  (:doc "Validates graph integrity for processes.")
370  (:body
371    (let [node-ids (meta/child-identifier-set input :node :id)
372          trigger-forms (meta/child-forms input :trigger)
373          trigger-kind (if (empty? trigger-forms)
374                         nil
375                         (let [trigger (first trigger-forms)]
376                           (meta/identifier trigger :kind)))
377          edges (meta/child-forms input :edge)]
378      (list/flat-map edges
379        (fn [edge]
380          (let [from-id (meta/identifier edge :from)
381                to-id (meta/identifier edge :to)]
382            (diag/concat
383              (if (if (set/contains? node-ids from-id)
384                    true
385                    (if (= from-id "start")
386                      true
387                      (= from-id trigger-kind)))
388                []
389                [(diag/error :form edge :message (str "unknown process node '" from-id "'"))])
390              (if (set/contains? node-ids to-id)
391                []
392                [(diag/error :form edge :message (str "unknown process node '" to-id "'"))]))))))))

570(meta-fn view/construct
571  (:kind construct)
572  (:input FormMetaInput)
573  (:output ViewConstruct)
574  (:doc "Constructs view IR from query and column definitions.")
575  (:body
576    (let [name (meta/declaration-name input)
577          description (meta/slot-string input :description)
578          query-ref (meta/slot-string input :query)
579          column-forms (meta/child-forms input :column)
580          column-values (meta/slot-string-list input :column)
581          default-sort-values (meta/slot-string-list input :default-sort)
582          default-sort (if (empty? default-sort-values)
583            nil
584            (let [field-name (first default-sort-values)
585                  direction-name (nth default-sort-values 1)]
586              (construct/object
587                :field field-name
588                :direction (if (= direction-name ":desc") "desc" "asc"))))
589          row-action (meta/slot-expr input :row-action)
590          ;; Extended ViewSpec: state declarations
591          state-forms (meta/child-forms input :state)
592          state-map (if (empty? state-forms) nil
593            (into (list/map state-forms
594              (fn [sf]
595                (let [state-kind (if (meta/slot-string sf :type)
596                                   (meta/slot-string sf :type)
597                                   "null")
598                      state-initial (meta/slot-value sf :initial)]
599                  [(meta/identifier sf :name)
600                 (construct/object
601                   :kind state-kind
602                   :type state-kind
603                   :initial state-initial)])))))
604          ;; Extended ViewSpec: input parameter declarations
605          input-param-forms (meta/child-forms input :input-param)
606          input-map (if (empty? input-param-forms) nil
607            (into (list/map input-param-forms
608              (fn [inf]
609                [(meta/identifier inf :name)
610                 (construct/object
611                   :type (meta/slot-string inf :type)
612                   :description (meta/slot-string inf :description))]))))
613          ;; Extended ViewSpec: named query bindings
614          nq-forms (meta/child-forms input :named-query)
615          queries-map (if (empty? nq-forms) nil
616            (into (list/map nq-forms
617              (fn [nq]
618                (let [params (view/compile-expr-record (meta/slot-expr nq :params))]
619                  [(meta/identifier nq :name)
620                   (construct/object
621                     :ref (meta/slot-string nq :ref)
622                     :queryRef (meta/slot-string nq :ref)
623                     :params params
624                     :dependsOn (meta/slot-string-list nq :depends-on))])))))
625          ;; ViewSpec reusable component defs
626          def-forms (meta/child-forms input :def)
627          defs-map (if (empty? def-forms) nil
628            (into (list/map def-forms
629              (fn [df]
630                [(meta/identifier df :name)
631                 (meta/compile-descriptor-tree "viewspec"
632                   (meta/slot-expr df :layout))]))))
633          ;; Extended ViewSpec: layout tree
634          layout-expr (meta/slot-expr input :layout)
635          layout-tree (if (nil? layout-expr) nil
636            (meta/compile-descriptor-tree "viewspec"
637              layout-expr))
