1; compiler.lisp
2; -----------------------------------------------------------------------------
3; Self-describing compiler prelude for the Lisp DSL.
4;
5; This file defines the meta-level forms that the compiler uses to understand
6; language constructs. These are compiler infrastructure, not ontology semantics.
7;
8; Forms defined here:
9;   define-form, meta-fn, syntax, identifier, slot, bind, validate, elaborate,
10;   construct-fn, bindings-fn, result-type-fn, validate-fn
11;
12; Plus: the descriptor-construct generic meta hook, worked examples of
13; meta-fn bodies, and the minimal canonical helper vocabulary spec.
14;
15; Any domain (ontology, hardware description, etc.) can define its own
16; forms using these meta constructs.
17;
18; How to read this file:
19; - A `define-form` declares one Lisp head and the shape of its arguments.
20; - `:identifiers` are positional values immediately after the head.
21; - `:slots` are keyword clauses such as `(:field ...)` or `(:returns ...)`.
22; - `:bindings-fn`, `:validate-fn`, `:construct-fn`, and `:result-type-fn`
23;   name compile-time hooks implemented in a companion compiler prelude.
24; - Most ontology authors do not write these forms directly. They explain
25;   how authoring forms like `define-entity` are recognized and lowered.
26;
27; Status: executable (parsed by lisp-v2, meta-fn bodies run at compile time)
28; -----------------------------------------------------------------------------

31; SECTION 1 — SELF-DESCRIBING META LAYER
32; =============================================================================
33;
34; These are the forms the ontology prelude would need in order to describe its
35; own forms. They are the "maximally needed" POC layer for self-description.
36;
37; Unclear point:
38; - whether these should literally be ordinary `(form ...)` declarations in the
39;   final system, or come from an even smaller kernel bootstrap seed.
40; - likely answer: tiny kernel seed bootstraps these, then these bootstrap the
41;   rest of ontology.lisp.

43(define-form form
44  (:doc "Defines a canonical DSL form descriptor.")
45  (:phase meta)
46  (:identifiers

48  (:slots

73; `meta-fn` is the bridge from declarative syntax descriptors to executable
74; compiler behavior. Domain preludes use it to say "when this form appears,
75; run this pure compile-time function to infer bindings, validate slots, choose
76; a result type, or construct IR."
77(define-form meta-fn
78  (:doc "Declares a pure compile-time meta hook.")
79  (:phase meta)
80  (:identifiers
81    [name Symbol {:declaration true}])
82  (:slots
83    [kind value {:required true}]
84    [input value {:required true}]
85    [output value {:required true}]
86    [capabilities value {:many true}]
87    [doc value]
88    [body expr {:required true}])
89  (:validate-fn meta-fn-validate)
90  (:construct-fn meta-fn-construct)
91  (:result-type-fn meta-fn-result-type))

93(define-form syntax
94  (:doc "Declares surface heads and aliases for a form.")
95  (:phase meta)
96  (:slots
97    [head value {:many true}]
98    [alias value {:many true}])
99  (:construct-fn descriptor-construct)
100  (:result-type (constant SyntaxDescriptor)))

102; Positional identifiers are the non-keyword arguments after a form head.
103; Example: in `(define-entity Employee ...)`, `Employee` is the `name`
104; identifier declared by the `define-entity` descriptor.
105(define-form identifier
106  (:doc "Declares a positional identifier for a form.")
107  (:phase meta)
108  (:identifiers
109    [name Symbol]
110    [kind Symbol])
111  (:slots
112    [declaration value]
113    [doc value])
114  (:construct-fn descriptor-construct)
115  (:result-type (constant IdentifierDescriptor)))

117; Slots are keyword clauses inside a form. A required slot must be present.
118; A many slot can appear more than once. Child-form metadata lets a repeated
119; slot parse compact nested forms such as `(:field [employee/name String])`.
120(define-form slot
121  (:doc "Declares a named slot for a form.")
122  (:phase meta)
123  (:identifiers
124    [name Symbol]
125    [kind Symbol])
126  (:slots
127    [required value]
128    [many value]
129    [alias value {:many true}]
130    [doc value]
131    [type value]
132    [type-from value])
133  (:construct-fn descriptor-construct)
134  (:result-type (constant SlotDescriptor)))

241; =============================================================================
242; SECTION 5 — MINIMAL CANONICAL HELPER VOCABULARY
243; =============================================================================
244;
245; This section lists the smallest helper surface I currently think the kernel +
246; meta substrate would need in order to support the worked examples above.
247;
248; Goal:
249; - enough power for full self-description and bootstrapping
250; - not enough power to become an unconstrained compiler plugin API
251;
252; -----------------------------------------------------------------------------
253; 5A. Input model helpers
254; -----------------------------------------------------------------------------
255;
256; These helpers expose structured views of the current compile-time input.
257; They should be preferred over direct access to raw normalized/compiler internals.
258;
259; Core meta/input helpers:
260; - (meta/descriptor input)
261; - (meta/form-name input)
262; - (meta/declaration-name input)
263; - (meta/loc input)
264; - (meta/normalized-form input)
265; - (meta/semantic-env input)
266;
267; Slot/identifier access:
268; - (meta/slot-value input :slot)
269; - (meta/slot-values input :slot)
270; - (meta/slot-string input :slot)
271; - (meta/slot-string-list input :slot)
272; - (meta/slot-symbol input :slot)
273; - (meta/slot-expr input :slot)
274; - (meta/slot-ref input :slot DeclarationKind)
275; - (meta/identifier input :name)
276;
277; Child-form traversal:
278; - (meta/child-forms input :slot)
279; - (meta/child-form input :slot)
280; - (meta/child-identifier-set input :slot :identifier)
281;
282; Generic declaration reflection:
283; - (meta/lookup-declaration input name)
284; - (meta/declaration-kind declaration)
285; - (meta/declaration-type declaration)
286; - (meta/declaration-field declaration field-name)
287; - (meta/project-type declaration [field ...])
288;
289; Unclear / maybe too low-level:
290; - exposing raw normalized maps or raw declaration indexes directly
291; - likely should stay internal to TypeScript
292;
293; -----------------------------------------------------------------------------
294; 5B. Type helpers
295; -----------------------------------------------------------------------------
296;
297; These helpers let meta functions describe HM results without constructing raw
298; TS runtime objects directly.
299;
300; Minimal type helpers:
301; - (type/unknown)
302; - (type/constant Name)
303; - (type/list T)
304; - (type/vector T)
305; - (type/record [field type] ...)
306; - (type/ref Name)
307; - (type/project-row row [field ...])
308;
309; Type-check query helpers:
310; - (meta/expr-assignable-to? input expr Type)
311; - maybe later: (meta/infer-expr-type input expr)
312;
313; -----------------------------------------------------------------------------
314; 5C. Binding helpers
315; -----------------------------------------------------------------------------
316;
317; These helpers construct validated binding payloads.
318;
319; Minimal binding helpers:
320; - (bindings/empty)
321; - (bindings/of [name Type] ...)
322; - (bindings/merge a b ...)
323;
324; Possibly useful later:
325; - (bindings/when condition payload)
326;
327; -----------------------------------------------------------------------------
328; 5D. Construct helpers
329; -----------------------------------------------------------------------------
330;
331; These helpers produce structured elaboration payloads.
332;
333; Minimal construct helpers:
334; - (construct/from-descriptor ...)
335; - (construct/object ...)
336;
337; -----------------------------------------------------------------------------
338; 5E. Diagnostic helpers
339; -----------------------------------------------------------------------------
340;
341; These helpers produce structured diagnostics without exposing TS internals.
342;
343; Minimal diagnostic helpers:
344; - (diag/error :message "...")
345; - (diag/error :slot :where :message "...")
346; - (diag/error :form child-form :message "...")
347; - (diag/concat a b ...)
348; - (diag/validate-membership-list values allowed :slot :slot-name)
349; - (diag/validate-default-in-list default values :slot :slot-name)
350;
351; Likely useful:
352; - (diag/warning ...)
353; - (diag/info ...)
354;
355; -----------------------------------------------------------------------------
356; 5F. Small safe collection/string helpers
357; -----------------------------------------------------------------------------
358;
359; These are probably the maximum general-purpose helpers we should allow.
360;
361; Collections:
362; - (empty? xs)
363; - (list/flat-map xs fn)
364; - (list/map xs fn)
365; - (list/filter xs fn)
366; - (set/contains? set value)
367;
368; Strings:
369; - (str a b c ...)
370;
371; Control flow:
372; - let
373; - if
374; - small local fn
375;
376; I would avoid for now:
377; - unrestricted recursion
378; - macros in meta-fn bodies
379; - arbitrary higher-order library imports
380; - mutation
381;
382; -----------------------------------------------------------------------------
383; 5G. Probably kernel-owned vs ontology-owned
384; -----------------------------------------------------------------------------
385;
386; Kernel-owned generic helpers:
387; - meta/input access primitives
388; - binding constructors
389; - type constructors
390; - construct protocol dispatch
391; - diagnostic constructors
392; - tiny safe collection/string helpers
393;
394; This separation is important.
395; The kernel should host DSLs; the ontology prelude should define ontology logic.
396;
397; -----------------------------------------------------------------------------
398; 5H. Bootstrap minimum
399; -----------------------------------------------------------------------------
400;
401; If we work backwards from this file, the smallest trusted bootstrap seed likely
402; needs to support:
403; - parsing S-expressions
404; - reading a tiny seed descriptor for `form`
405; - reading a tiny seed descriptor for `meta-fn`
406; - validating structured helper calls in meta-fn bodies
407; - executing constrained meta hooks
408;
409; After that, ontology.lisp can define the rest of its own surface.
410;
