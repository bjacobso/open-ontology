1; viewspec-compiler.lisp
2; -----------------------------------------------------------------------------
3; Hosted ViewSpec compiler hooks.
4;
5; These hooks implement validation behavior for ontology forms whose semantics
6; depend on the hosted ViewSpec DSL.
7; -----------------------------------------------------------------------------
8 
9(meta-fn view/validate
10  (:kind validate)
11  (:input FormMetaInput)
12  (:output DiagnosticList)
13  (:doc "Type-checks the view layout tree against the ViewSpec component catalog.")
14  (:body
15    (let [layout-expr (meta/slot-expr input :layout)]
16      (let [state-forms (meta/child-forms input :state)
17            state-info (into (list/map state-forms
18              (fn [sf]
19                [(meta/identifier sf :name)
20                 (if (meta/slot-string sf :type)
21                   (meta/slot-string sf :type)
22                   "any")])))
23            main-query (meta/slot-string input :query)
24            nq-forms (meta/child-forms input :named-query)
25            nq-names (list/map nq-forms
26              (fn [nq] (meta/identifier nq :name)))
27            all-query-names (if (nil? main-query)
28              nq-names
29              (conj nq-names main-query))
30            input-forms (meta/child-forms input :input-param)
31            input-names (list/map input-forms
32              (fn [inf] (meta/identifier inf :name)))
33            def-forms (meta/child-forms input :def)
34            def-names (list/map def-forms
35              (fn [df] (meta/identifier df :name)))]
36        (if (nil? layout-expr)
37          []
38          (meta/validate-descriptor-tree
39            "viewspec"
40            layout-expr
41            state-info
42            all-query-names
43            input-names
44            def-names))))))
