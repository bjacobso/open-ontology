1; ontology.lisp
2; -----------------------------------------------------------------------------
3; Domain ontology prelude — defines the ontology language constructs.
4;
5; This file uses the compiler's meta forms (define-form, meta-fn) to define
6; the domain-specific constructs that ontology authors use.
7;
8; See ontology-compiler.lisp for the elaboration hooks (meta-fn implementations).
9;
10; Forms defined here:
11;
12; HTTP APIs:  define-schema, define-error, define-api-group, endpoint, param
13; Structure:  define-entity, define-relation, define-record, define-link
14; Queries:    define-query, define-view
15; Navigation: define-workspace
16; Compliance: define-constraint, resolution
17; Operations: define-action, define-mutation
18; Processes:  define-process, trigger, node, edge, guard
19; Documents:  define-document, page, field, completion-mutation, option
20; L10n:       define-document-locale, role, section, locale-field, define-document-localized
21; PDF:        define-pdf-mapping, direct, computed, switch, case, set
22;
23; Depends on: compiler.lisp (define-form, meta-fn, etc.)
24;
25; Authoring rules for humans and agents:
26; - Use the canonical `define-*` forms. Older forms such as `def-entity`,
27;   `def-action`, `def-rule`, bare `entity`, and bare `query` are legacy
28;   spellings and should be rewritten.
29; - Field names are usually namespace-qualified symbols such as
30;   `employee/name`, which lower to runtime attributes like `:employee/name`.
31; - Type references use symbols (`String`, `Bool`, `Employee`) or compound
32;   type forms such as `(Ref Department)`.
33; - Runtime behavior for `:do`, `:where`, and ViewSpec bindings is ordinary
34;   Lisp checked by the compiler hooks in ontology-compiler.lisp.
35;
36; Minimal authoring example:
37;
38;   (define-entity Employee
39;     (:field [employee/name String {:required true}])
40;     (:field [employee/department (Ref Department)]))
41;
42;   (define-record "emp:alice" Employee
43;     (:field [employee/name "Alice"]))
44;
45;   (define-query employees
46;     (:from Employee)
47;     (:select [employee/name employee/department]))
48;
49; Status: executable (parsed by lisp-v2, meta-fn bodies run at compile time)
50; -----------------------------------------------------------------------------

352; Entity definitions declare the shape of facts that can be asserted for a
353; type. The `field` slot is repeated and normally uses vector shorthand:
354;   (:field [employee/name String {:required true}])
355; The first vector item is the attribute name, the second is the type, and the
356; optional map carries field metadata.
357(define-form define-entity
358  (:phase domain)
359  (:doc "Canonical entity/schema declaration.")
360  (:identifiers
361    (identifier name Symbol (:declaration true)))
362  (:slots
363    (slot doc value)
364    (slot role value)
365    (slot id-pattern value)
366    (slot field value
367      (:many true)
368      (:required true)
369      (:child-form field)
370      (:child-identifier name Value)
371      (:child-slot type expr (:positional true))
372      (:child-slot required value)
373      (:child-slot indexed value)))
374  (:bindings
375    (bind bind-declaration-name (:identifier name) (:type SchemaDecl)))
376  (:bindings-fn entity/bindings)
377  (:extensions
378    (:artifact
379      (:payload (:contract EntityPayload))))
380  (:construct-fn entity/construct)
381  (:construct
382    [kind "Entity"]
383    [name (or declaration-name "anonymous-entity")]
384    [fields (entity-fields field)]
385    [loc loc])
386  (:declaration-type (row))
387  (:result-type (constant SchemaDecl)))
