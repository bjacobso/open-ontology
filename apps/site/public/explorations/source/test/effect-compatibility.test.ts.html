1import { Effect, Result, Schema } from "effect";
2import { DatabaseId, EntityId, KvTriples, TransactionId, TripleId, Triples, string, type DatalogQuery } from "@triplex-build/triplex";
3import { describe, expect, it } from "vitest";
4import { Change, Input, Ontology, Property, defineOntology, makeOntologyRuntime, materializeOntology, ontologyToIR, toTriplexConfig } from "../src/index.js";
5 
6describe("Effect 4 compatibility", () => {
7  it.each([Schema.Date, Schema.NumberFromString])("rejects nonportable action inputs: %s", (schema) => {
8    const action = Ontology.ActionType("nonportable", {
9      input: Schema.Struct({ value: schema }),
10      changes: [],
11    });
12    expect(() => ontologyToIR(defineOntology({ name: "invalid" }, [action])))
13      .toThrow("cannot be restored from portable JSON Schema");
14  });
15 
16  it.each([
17    ["non-Unicode patterns", Schema.String.check(Schema.isPattern(/^\S/))],
18    ["UTF-16 length approximations", Schema.String.check(Schema.isMinLength(2))],
19  ] as const)("rejects approximate JSON Schema checks: %s", (_name, schema) => {
20    const action = Ontology.ActionType("approximate", {
21      input: Schema.Struct({ value: schema }),
22      changes: [],
23    });
24    expect(() => ontologyToIR(defineOntology({ name: "invalid" }, [action])))
25      .toThrow("cannot be reflected to JSON Schema");
26  });
27 
28  it("preserves Triplex branded-id constructors and schema validation", () => {
29    const cases: readonly (readonly [
30      Schema.Schema<string> & { readonly DecodingServices: never; readonly make: (value: string) => string; readonly decode: (value: string) => Effect.Effect<string, Schema.SchemaError> },
31      string,
32      string,
33    ])[] = [
34      [EntityId, "item:é", ""],
35      [DatabaseId, "local-db", "INVALID"],
36      [TripleId, "01ARZ3NDEKTSV4RRFFQ69G5FAV", "short"],
37      [TransactionId, "_tx/01ARZ3NDEKTSV4RRFFQ69G5FAV", "invalid"],
38    ];
39    for (const [schema, valid, invalid] of cases) {
40      expect(schema.make(valid)).toBe(valid);
41      expect(Effect.runSync(schema.decode(valid))).toBe(valid);
42      expect(Schema.decodeUnknownSync(schema)(valid)).toBe(valid);
43      expect(() => schema.make(invalid)).toThrow();
44      expect(Result.isFailure(Effect.runSync(Effect.result(schema.decode(invalid))))).toBe(true);
45    }
46  });
47 
48  it("restores portable action checks and executes Triplex configuration and actions", async () => {
49    const item = Ontology.ObjectType("Item", {
50      properties: { title: Property.make(":item/title", Schema.String).required() },
51    });
52    const action = Ontology.ActionType("create-item", {
53      input: Schema.Struct({
54        id: Schema.String,
55        title: Schema.String.check(Schema.isMinCodePoints(2), Schema.isPattern(/^\S/u)),
56        priority: Schema.Number.check(Schema.isInt(), Schema.isGreaterThan(0)),
57        note: Schema.optional(Schema.String),
58      }),
59      changes: [Change.create(item, Input.value("id"), { ":item/title": Input.value("title") })],
60    });
61    const ir = ontologyToIR(defineOntology({ name: "compatibility" }, [item, action]));
62    const restored = materializeOntology(JSON.parse(JSON.stringify(ir)));
63    expect(ontologyToIR(restored)).toEqual(ir);
64    const runtime = makeOntologyRuntime(restored);
65    const result = await Effect.runPromise(Effect.gen(function* () {
66      const nodes = yield* toTriplexConfig(ir).nodes;
67      const valid = { id: "item:é", title: "Café", priority: 1 };
68      for (const input of [
69        { ...valid, title: "x" },
70        { ...valid, title: " space" },
71        { ...valid, priority: 0 },
72        { ...valid, priority: 1.5 },
73        { ...valid, note: 42 },
74      ]) {
75        expect(Result.isFailure(yield* Effect.result(runtime.invoke("create-item", input)))).toBe(true);
76      }
77      yield* runtime.invoke("create-item", valid);
78      const triples = yield* Triples;
79      return { nodes, facts: yield* triples.entity(EntityId.make(valid.id)) };
80    }).pipe(Effect.provide(KvTriples.layer)));
81    expect(result.nodes.length).toBeGreaterThan(1);
82    expect(result.facts).toHaveLength(1);
83    expect(result.facts[0]?.value).toEqual(string("Café"));
84  });
85 
86  it("continues query and entity cursors at their original snapshot and rejects invalid cursors", async () => {
87    await Effect.runPromise(Effect.gen(function* () {
88      const triples = yield* Triples;
89      yield* triples.transact(["a", "b", "c"].map((id) => ({
90        op: "assert" as const,
91        entityId: EntityId.make(`item:${id}`),
92        entityType: "Item",
93        attribute: ":item/title",
94        value: string(`Café ${id}`),
95      })));
96      const query: DatalogQuery = { find: ["?title"], where: [["?item", ":item/title", "?title"]] };
97      const firstQuery = yield* triples.query(query, { pageSize: 1 });
98      const firstEntities = yield* triples.entityPage({ entityType: "Item", limit: 1 });
99      expect(firstQuery.nextCursor).toEqual(expect.any(String));
100      expect(firstEntities.nextCursor).toEqual(expect.any(String));
101      // A later commit must not enter either continuation's captured snapshot.
102      yield* triples.transact([{
103        op: "assert", entityId: EntityId.make("item:bb"), entityType: "Item",
104        attribute: ":item/title", value: string("Café bb"),
105      }]);
106      const secondQuery = yield* triples.query(query, { pageSize: 1, cursor: firstQuery.nextCursor! });
107      const thirdQuery = yield* triples.query(query, { pageSize: 1, cursor: secondQuery.nextCursor! });
108      expect([firstQuery, secondQuery, thirdQuery].flatMap((page) => page.results)).toEqual([
109        { "?title": "Café a" }, { "?title": "Café b" }, { "?title": "Café c" },
110      ]);
111      expect(thirdQuery.nextCursor).toBeUndefined();
112      const remainingEntities = yield* triples.entityPage({ entityType: "Item", limit: 2, cursor: firstEntities.nextCursor! });
113      expect(remainingEntities.entities.map((facts) => facts[0]?.entityId)).toEqual(["item:b", "item:c"]);
114      expect(remainingEntities.snapshot).toEqual(firstEntities.snapshot);
115      expect(remainingEntities.nextCursor).toBeUndefined();
116      const malformedQuery = yield* Effect.result(triples.query(query, { cursor: "%%%" }));
117      const malformedEntities = yield* Effect.result(triples.entityPage({ entityType: "Item", cursor: "%%%" }));
118      const wrongEntityType = yield* Effect.result(triples.entityPage({ entityType: "Other", cursor: firstEntities.nextCursor! }));
119      const wrongQuery = yield* Effect.result(triples.query({ find: ["?item"], where: query.where }, { pageSize: 1, cursor: firstQuery.nextCursor! }));
120      for (const result of [malformedQuery, malformedEntities]) {
121        expect(result).toMatchObject({ _tag: "Failure", failure: { _tag: "PaginationCursorError", reason: "malformed" } });
122      }
123      for (const result of [wrongEntityType, wrongQuery]) {
124        expect(result).toMatchObject({ _tag: "Failure", failure: { _tag: "PaginationCursorError", reason: "query_mismatch" } });
125      }
126    }).pipe(Effect.provide(KvTriples.layer)));
127  });
128});
