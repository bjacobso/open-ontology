8type View = Readonly<Record<string, unknown>>;
9const numeric = (x: unknown): number => {
10  if (typeof x !== "number" || !Number.isFinite(x)) throw new TypeError("Expected a finite number");
11  return x;
12};
13const instant = (x: unknown): number => numeric(typeof x === "string" ? Date.parse(x) : x);
14const list = (x: unknown): readonly unknown[] => {
15  if (!Array.isArray(x)) throw new TypeError("Expected a list");
16  return x;
17};

19/** Interpreter for canonical RuntimeExpr data; no eval, host calls, or Forma compiler workarounds. */
20export const evaluate = (expression: unknown, scope: View): unknown => {
21  if (expression !== null && typeof expression === "object" && !Array.isArray(expression)) {
22    const value = expression as Record<string, unknown>;
23    if (value["kind"] === "raw-expr") return evaluate(value["expr"], scope);
24    if (value["$forma.runtimeExpr"] === "string-literal") return value["value"];
25    throw new TypeError("Unknown expression object");
26  }
27  if (typeof expression === "string") {
28    if (expression.startsWith(":")) return expression.slice(1);
29    return expression.split(".").reduce<unknown>((v, key) =>
30      v !== null && typeof v === "object" ? (v as View)[key] : undefined, scope);
31  }
32  if (!Array.isArray(expression)) return expression;
33  if (expression.length === 0 || typeof expression[0] !== "string" || expression[0].startsWith(":")) {
34    return expression.map((x) => evaluate(x, scope));
35  }
36  const [op, ...args] = expression as [string, ...unknown[]];
37  const at = (i: number) => evaluate(args[i], scope);
38  switch (op) {
39    case "and": return args.every((x) => evaluate(x, scope) === true);
40    case "or": return args.some((x) => evaluate(x, scope) === true);
41    case "not": return at(0) !== true;
42    case "=": case "is?": return at(0) === at(1);
43    case "known?": return at(0) !== undefined && at(0) !== null;
44    case "in?": return list(at(1)).includes(at(0));
45    case "before?": return instant(at(0)) < instant(at(1));
46    case "after?": return instant(at(0)) > instant(at(1));
47    case "days-between": return Math.floor((instant(at(1)) - instant(at(0))) / 86_400_000);
48    case "sum": return list(at(0)).reduce<number>((sum, x) => sum + numeric(x), 0);
49    case "+": return args.reduce<number>((sum, x) => sum + numeric(evaluate(x, scope)), 0);
50    case "*": return args.reduce<number>((product, x) => product * numeric(evaluate(x, scope)), 1);
51    case "-": return args.length === 1 ? -numeric(at(0)) : numeric(at(0)) - numeric(at(1));
52    case "/": return numeric(numeric(at(0)) / numeric(at(1)));
53    case "map": case "filter": {
54      const fn = args[0];
55      if (!Array.isArray(fn) || fn[0] !== "fn" || !Array.isArray(fn[1]) || typeof fn[1][0] !== "string") {
56        throw new TypeError(`${op} expects a single-argument fn`);
57      }
58      const parameter = fn[1][0];
59      const apply = (x: unknown) => evaluate(fn[2], { ...scope, [parameter]: x });
60      return op === "map" ? list(at(1)).map(apply) : list(at(1)).filter((x) => apply(x) === true);
61    }
62    default: throw new TypeError(`Unsupported Onto expression: ${op}`);
63  }
64};

72/** Guard every observed entity (including absence) and pass preconditions at the Triples boundary. */
73export const makeStoryHost = (declarations: readonly ElaboratedDeclaration[], ir: OntologyIR) => {
74  const runtime = makeOntologyRuntime(ir);
75  const nouns = nounsOf(declarations);
76  const steps = stepsOf(declarations);
77  return {
78    runtime,
79    invoke: (name: string, input: View, options: { readonly actor: string; readonly commandId: string; readonly now: number },
80      beforeCommit?: Effect.Effect<void, never, Triples>) => Effect.gen(function* () {
81      const step = steps.find((s) => s.name === name)?.step;
82      if (!step) throw new TypeError(`Unknown story step ${name}`);
83      const triples = yield* Triples;
84      const snapshots = new Map<string, readonly Triple[]>();
85      const read = (id: string) => Effect.gen(function* () {
86        const cached = snapshots.get(id);
87        if (cached) return cached;
88        const facts = yield* triples.entity(EntityId.make(id));
89        snapshots.set(id, facts);
90        return facts;
91      });
92      const view = (owner: string, id: string, depth: number, initial?: View): Effect.Effect<View | undefined, unknown> =>
93        Effect.gen(function* () {
94          const facts = yield* read(id);
95          if (!initial && facts.length === 0) return undefined;
96          const noun = nouns.find((n) => n.name === owner)!;
97          const record: Record<string, unknown> = {};
98          for (const field of noun.fields) {
99            const stored = facts.filter((f) => f.attribute === attribute(owner, field.name)).map((f) => f.value.value);
100            const values = initial && field.name in initial
101              ? field.many ? list(initial[field.name]) : [initial[field.name]] : stored;
102            const resolved: unknown[] = [];
103            for (const value of values) {
104              resolved.push(field.link && depth > 0 ? yield* view(field.link, String(value), depth - 1) : value);
105            }
106            record[field.name] = field.many ? resolved : resolved[0];
107          }
108          return record;
109        });
110      const id = String(input["id"]);
111      const current = yield* read(id);
112      const refuse = (reason: string): StepResult => ({ accepted: false, reason, commandId: options.commandId });
113      const noun = nouns.find((n) => n.name === step.object)!;
114      // Check terminal states before creation/guard checks, including attempts to reuse a create step.
115      for (const set of step.sets) {
116        const field = noun.fields.find((f) => f.name === set.field)!;
117        const value = current.find((f) => f.attribute === attribute(noun.name, field.name))?.value.value;
118        if (field.final.includes(String(value)) && value !== set.value) return refuse(`final ${noun.name}.${field.name}: ${value}`);
119      }
120      if (step.creates && current.length > 0) return refuse("record already exists");
121      if (!step.creates && current.length === 0) return refuse("record does not exist");
122      const record = yield* view(step.object, id, 3, step.creates ? input : undefined);
123      const scope = { [step.object.toLowerCase()]: record, now: options.now };
124      const failed = step.requires.find((e: Expression) => evaluate(e, scope) !== true);
125      if (failed) return refuse(`requires ${JSON.stringify(failed.expr)}`);
126      const preconditions: readonly TransactionPrecondition[] = [...snapshots].map(([entityId, facts]) => ({
127        _tag: "EntityState", entityId: EntityId.make(entityId), tripleIds: facts.map((f) => f.id),
128      }));
129      if (beforeCommit) yield* beforeCommit;
130      // invoke() does not forward preconditions. A scoped service decorator adds them to its actual write.
131      const receipt = yield* runtime.invoke(name, input, options).pipe(Effect.provideService(Triples, {
132        ...triples,
133        transact: (operations, meta) => triples.transact(operations, {
134          ...meta, preconditions: [...(meta?.preconditions ?? []), ...preconditions],
135        }),
136      }), Effect.result);
137      if (receipt._tag === "Failure") {
138        if (receipt.failure._tag === "TransactionConflictError") return refuse("concurrent entity change; retry from fresh state");
139        return yield* Effect.fail(receipt.failure);
140      }
141      return { accepted: true, commandId: options.commandId } satisfies StepResult;
142    }),
143  };
144};
