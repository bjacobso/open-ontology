1; onto.lisp: an OntoLang-shaped prelude for Forma.
2;
3; Seven forms: project, actor, object, rule, formula, story (with step children), seed.
4; Every expression hole (requires, rule, formula) is type-checked by Forma's inference
5; against a record view of the declared object, so reservation.room-class.rate is
6; checked by type, not by string lookup.
7 
8; Forma keyword-in-lambda workaround: monomorphic is? (upstream fix: https://github.com/bjacobso/forma/pull/110).
9; ── Vocabulary: what every condition and formula may call ──
10(type Instant (Brand String))
11; (One X) is a link to an X; (Many X) is X[] in OntoLang.
12(type (One a) a)
13(type (Many a) (List a))
14(define
15  onto/vocabulary
16  {:= (quote (-> a a Bool))
17   :is? (quote (-> Keyword Keyword Bool))
18   :now (quote Instant)
19   :before? (quote (-> Instant Instant Bool))
20   :after? (quote (-> Instant Instant Bool))
21   :days-between (quote (-> Instant Instant Int))
22   :known? (quote (-> a Bool))
23   :in? (quote (-> a (List a) Bool))
24   :sum (quote (-> (List Int) Int))})

26; ── Helpers over authored data ──
27(define name-of [x] (if (nil? x) "" (if (keyword? x) (keyword/name x) (str x))))
28(define same? [a b] (= (name-of a) (name-of b)))
29(define member? [x xs] (some (fn [y] (same? x y)) xs))
30(define pairs [r] (if (nil? r) [] (if (map? r) (meta/entries r) r)))
31(define lookup [record k] (let [hit (first (filter (fn [[f t]] (same? f k)) (pairs record)))] (if (nil? hit) nil (nth hit 1))))
32(define head-name [x] (if (and (list? x) (not (empty? x))) (name-of (first x)) ""))
33(define diag [code message slot] {:severity :error :code code :message message :slot slot})
34(define warn [code message slot] {:severity :warning :code code :message message :slot slot})
35 
36; (One X) is a link, (Many X) is X[] in OntoLang, (Union :a :b) is a | b.
37(define link-target [ty] (if (or (= (head-name ty) "One") (= (head-name ty) "Many")) (nth ty 1) nil))
38(define many? [ty] (= (head-name ty) "Many"))
39(define states-of [ty] (if (= (head-name ty) "Union") (rest ty) nil))
40(define declared-noun? [x] (or (member? x (declarations-of (quote object))) (member? x (declarations-of (quote actor)))))

42; Formulas show up on their object's view as if they were fields.
43(define
44  formulas-for
45  [owner]
46  (filter (fn [f] (same? (declaration-hole f :of) owner)) (declarations-of (quote formula))))

48; What a reference to an object looks like to an expression: a record of its fields
49; and formulas, following links `depth` levels deep.
50(define
51  view-of
52  [owner depth]
53  (merge
54    (reduce
55      (fn
56        [acc [k ty]]
57        (assoc acc (keyword (name-of k)) (lift ty depth)))
58      {}
59      (pairs (declaration-fields owner)))
60    (reduce
61      (fn [acc f] (assoc acc (keyword (name-of (declaration-hole f :field))) (declaration-hole f :returns)))
62      {}
63      (formulas-for owner))))
64(define
65  lift
66  [ty depth]
67  (let
68    [target (link-target ty)]
69    (if (not (nil? (states-of ty)))
70      (quote Keyword)
71      (if (nil? target)
72        ty
73        (let
74          [v (if (> depth 0) (view-of target (- depth 1)) (quote String))]
75          (if (many? ty) [(sym "List") v] v))))))
76(define
77  scope-for
78  [owner]
79  (merge onto/vocabulary {(keyword (lower (name-of owner))) (view-of owner 3)}))

81; Follow a dotted path from an object: (path-type Reservation ["room-class" "rate"]) → Int.
82(define
83  path-type
84  [owner parts]
85  (let
86    [ty (lookup (declaration-fields owner) (first parts))]
87    (if (or (nil? ty) (= (count parts) 1))
88      ty
89      (let [target (link-target ty)] (if (nil? target) nil (path-type target (rest parts)))))))
90; (is? reservation.status :confirmd): the keyword must be one of that field's states.
91(define
92  states/check
93  [owner e slot]
94  (if (and (list? e) (= (head-name e) "is?") (= (count e) 3) (symbol? (nth e 1)) (keyword? (nth e 2)))
95    (let
96      [parts (split (name-of (nth e 1)) ".")
97       ty (if (same? (first parts) (lower (name-of owner))) (path-type owner (rest parts)) nil)
98       states (if (nil? ty) nil (states-of ty))]
99      (if (or (nil? states) (member? (nth e 2) states))
100        []
101        [(diag "onto/unknown-state"
102           (str ":" (name-of (nth e 2)) " is not a state of " (name-of (nth e 1)) "; it is one of "
103                (reduce (fn [acc x] (str acc (if (= acc "") "" ", ") ":" (name-of x))) "" states) ".")
104           slot)]))
105    (if (list? e) (flat-map (fn [x] (states/check owner x slot)) e) [])))

116; ── actor / object ──
117(type FieldIR {:name String :type Syntax :link (Option Symbol) :many Bool :states (Option (List Keyword)) :final (List Keyword)})
118(define
119  fields->ir
120  [fields final]
121  (map
122    (fn
123      [[k ty]]
124      {:name (name-of k)
125       :type ty
126       :link (link-target ty)
127       :many (many? ty)
128       :states (states-of ty)
129       :final (filter (fn [s] (member? s (pairs (states-of ty)))) (pairs (lookup final k)))})
130    (pairs fields)))
131(define
132  links/check
133  [{:keys [fields]}]
134    (flat-map
135      (fn
136        [[k ty]]
137        (let
138          [target (link-target ty)]
139          (if (and (not (nil? target)) (not (declared-noun? target)))
140            [(diag "onto/unknown-link" (str (name-of k) " links to " (name-of target) ", which is not an actor or object.") :fields)]
141            [])))
142      (pairs fields)))
143(define
144  finals/check
145  [{:keys [fields final]}]
146    (flat-map
147      (fn
148        [[k finals]]
149        (let
150          [ty (lookup fields k)]
151          (if (nil? (states-of ty))
152            [(diag "onto/final-not-states" (str (name-of k) " is not a list of states, so it has no final ones.") :final)]
153            (flat-map
154              (fn
155                [s]
156                (if (member? s (states-of ty))
157                  []
158                  [(diag "onto/unknown-state" (str ":" (name-of s) " is not a state of " (name-of k) ".") :final)]))
159              finals))))
160      (pairs final)))

162(type ActorIR {:kind "Actor" :name Symbol :actor-type Keyword :doc String :joins Keyword :fields (List FieldIR)})
163(form
164  (actor name actor-type doc {:keys [joins]} fields)
165  "Someone, or something, that takes part in the work."
166  :types {:name (Declares Actor) :actor-type (Union :person :system) :doc String
167          :joins (Option (Union :sign-up :invite :none)) :fields (Record Type)}
168  :check links/check
169  :ir ActorIR
170  {:kind "Actor" :name name :actor-type actor-type :doc doc
171   :joins (if (nil? joins) (if (= actor-type :system) :none :invite) joins)
172   :fields (fields->ir fields {})})

174(type ObjectIR {:kind "Object" :name Symbol :doc String :fields (List FieldIR)})
175(form
176  (object name doc {:keys [final]} fields)
177  "A record the business keeps track of."
178  :types {:name (Declares Object) :doc String :final (Option Syntax) :fields (Record Type)}
179  :check (fn [holes] (concat (links/check holes) (finals/check holes)))
180  :ir ObjectIR
181  {:kind "Object" :name name :doc doc :fields (fields->ir fields (if (nil? final) {} final))})

183; ── rule: holds whichever story runs ──
184(type RuleIR {:kind "Rule" :name Symbol :of Symbol :doc String :holds RuntimeExpr})
185(form
186  (rule name of doc holds)
187  "Something that must always be true of every record of an object."
188  :types {:name (Declares Rule) :of (Refers Object) :doc String :holds (Expr Bool)}
189  :scope {:holds (fn [{:keys [of]}] (scope-for of))}
190  :check (fn [{:keys [of holds]}] (states/check of holds :holds))
191  :ir RuleIR
192  {:kind "Rule" :name name :of of :doc doc :holds holds})

211(define state-field [object] (first (filter (fn [[k ty]] (not (nil? (states-of ty)))) (pairs (declaration-fields object)))))
212(define field-type [object field] (lookup (declaration-fields object) field))

214; Every keyword-ish field name an expression mentions (reservation.status → :status).
215(define
216  flatten-keys
217  [e]
218  (if (symbol? e)
219    (map (fn [p] (keyword p)) (rest (split (name-of e) ".")))
220    (if (list? e) (flat-map flatten-keys e) [])))
221 
222(define
223  step/check
224  [{:keys [actor verb object to for new requires sets]}]
225  (concat
226    (if (and (not (nil? to)) (not (nil? for)))
227      [(diag "onto/two-counterparts" "A step says :to or :for, not both." :for)]
228      [])
229    (flat-map
230      (fn
231        [[field value]]
232        (let
233          [ty (field-type object field)]
234          (if (nil? ty)
235            [(diag "onto/unknown-field" (str (name-of object) " has no field " (name-of field) ".") :sets)]
236            (if (nil? (states-of ty))
237              []
238              (if (member? value (states-of ty))
239                ; A step that moves a status without saying what it moves *from* could
240                ; move a record out of a final state.
241                (if (or new (some (fn [r] (member? (keyword (name-of field)) (flatten-keys r))) (pairs requires)))
242                  []
243                  [(warn "onto/unguarded-transition"
244                     (str (name-of actor) " " (name-of verb) " " (name-of object) " sets " (name-of field) " to :" (name-of value)
245                          " without requiring its current state, so it could leave a final state. Add a requires on "
246                          (lower (name-of object)) "." (name-of field) ", or :new true if this step creates the record.")
247                     :sets)])
248                [(diag "onto/unknown-state" (str ":" (name-of value) " is not a state of " (name-of object) "." (name-of field) ".") :sets)])))))
249      (pairs sets))))
250(form
251  (story-child/step actor verb object {:keys [to for new requires sets]})
252  "One sentence of a story: who does what, to what."
253  :types {:actor (Refers Actor) :verb Symbol :object (Refers Object)
254          :to (Option (Refers Actor)) :for (Option (Refers Actor)) :new (Option Bool)
255          :requires (Option (List (Expr Bool))) :sets (Option Syntax)}
256  :scope {:requires (fn [{:keys [object]}] (scope-for object))}
257  :check (fn [holes] (concat (step/check holes) (states/check (get holes :object) (get holes :requires) :requires)))
258  :ir StepIR
259  {:kind "Step" :actor actor :verb verb :object object
260   :counterpart (if (nil? to) for to)
261   :creates (if (nil? new) false new)
262   :requires (if (nil? requires) [] requires)
263   :sets (map (fn [[k v]] {:field (name-of k) :value v}) (pairs sets))})

273; ── seed: the records a new app starts with ──
274(type SeedIR {:kind "Seed" :of Symbol :rows Syntax})
275(form
276  (seed of rows)
277  "Starting records for an object."
278  :types {:of (Refers Object) :rows Syntax}
279  :check
280    (fn
281      [{:keys [of rows]}]
282      (flat-map
283        (fn
284          [row]
285          (flat-map
286            (fn [[k v]] (if (nil? (field-type of k)) [(diag "onto/unknown-field" (str (name-of of) " has no field " (name-of k) ".") :rows)] []))
287            (pairs row)))
288        rows))
289  :ir SeedIR
290  {:kind "Seed" :of of :rows rows})
