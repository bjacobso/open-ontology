50export const kebab = (name: string): string => name.replace(/([a-z0-9])([A-Z])/g, "$1-$2").toLowerCase();
51export const attribute = (owner: string, field: string): `:${string}/${string}` => `:${kebab(owner)}/${field}`;
52export const declarationsOf = (declarations: readonly ElaboratedDeclaration[]): readonly Declaration[] =>
53  declarations.map(({ payload }) => payload as unknown as Declaration);
54export const nounsOf = (declarations: readonly ElaboratedDeclaration[]): readonly Noun[] =>
55  declarationsOf(declarations).filter((d): d is Noun => d.kind === "Actor" || d.kind === "Object");
56export const stepsOf = (declarations: readonly ElaboratedDeclaration[]) =>
57  declarationsOf(declarations).flatMap((d) => d.kind === "Story"
58    ? d.steps.map((step) => ({ name: `${kebab(d.name)}-${step.verb}`, step })) : []);

60const valueType = (field: Field): OntologyValueTypeIR => {
61  if (field.link) return { kind: "ref", target: field.link };
62  if (field.states || field.type === "String") return { kind: "string" };
63  if (field.type === "Int") return { kind: "number" };
64  if (field.type === "Instant") return { kind: "datetime" };
65  if (field.type === "Bool") return { kind: "boolean" };
66  throw new TypeError(`Unsupported Onto field type: ${JSON.stringify(field.type)}`);
67};
68const property = (owner: string, field: Field): PropertyIR => ({
69  name: field.name, key: attribute(owner, field.name), valueType: valueType(field),
70  required: false, cardinality: field.many ? "many" : "one", unique: false,
71});

73const lowerStep = (name: string, step: Step, nouns: readonly Noun[]): ActionTypeIR => {
74  const noun = nouns.find((n) => n.name === step.object);
75  if (!noun) throw new TypeError(`Unknown step object ${step.object}`);
76  // Create inputs supply the single-valued facts. Collection membership remains explicit facts.
77  const fields = step.creates ? noun.fields.filter((f) => !f.many && !step.sets.some((s) => s.field === f.name)) : [];
78  const id = { kind: "input" as const, name: "id" };
79  return {
80    kind: "action-type", name,
81    input: [{ name: "id", valueType: { kind: "string" } },
82      ...fields.map((f) => ({ name: f.name, valueType: valueType(f) }))],
83    changes: step.creates ? [{
84      kind: "create-object", objectType: step.object, id,
85      properties: Object.fromEntries([
86        ...fields.map((f) => [attribute(step.object, f.name), { kind: "input", name: f.name }]),
87        ...step.sets.map((s) => [attribute(step.object, s.field), s.value]),
88      ]),
89    }] : step.sets.flatMap((s) => [
90      { kind: "clear-property" as const, object: id, property: attribute(step.object, s.field) },
91      { kind: "set-property" as const, object: id, property: attribute(step.object, s.field), value: s.value },
92    ]),
93  };
94};

105/** Pure third-party language IR → portable ontology IR. No reads or writes. */
106export const lowerOnto = (declarations: readonly ElaboratedDeclaration[]): OntologyIR => {
107  const ds = declarationsOf(declarations);
108  const project = ds.find((d) => d.kind === "Project");
109  if (!project) throw new TypeError("An Onto model needs a project");
110  const nouns = nounsOf(declarations);
111  return {
112    kind: "ontology", name: project.name, version: "1",
113    objectTypes: nouns.map((n) => ({ kind: "object-type", name: n.name, description: n.doc,
114      properties: n.fields.map((f) => property(n.name, f)) })),
115    linkTypes: [],
116    actionTypes: stepsOf(declarations).map(({ name, step }) => lowerStep(name, step, nouns)),
117    queryTypes: ds.flatMap((d): readonly QueryTypeIR[] => {
118      if (supportsRule(d)) return [{
119        kind: "query-type", name: "check-out-after-check-in",
120        description: "Reservations satisfying the date-order rule; observation only, not invariant enforcement.",
121        query: { find: ["?reservation"], where: [
122          ["?reservation", ":reservation/check-in", "?checkIn"],
123          ["?reservation", ":reservation/check-out", "?checkOut"], [">", "?checkOut", "?checkIn"],
124        ] },
125      }];
126      if (supportsFormula(d)) return [{
127        kind: "query-type", name: "service-charges",
128        description: "Sum of fulfilled service charges by reservation; absent groups mean zero.",
129        query: {
130          find: ["?reservation", "?total"],
131          where: [["?reservation", ":reservation/services", "?service"],
132            ["?service", ":service/status", "fulfilled"], ["?service", ":service/charge", "?charge"]],
133          aggregate: [["sum", "?charge", "?total"]],
134        },
135      }];
136      return [];
137    }),
138  };
139};

142export const notLowered = (declarations: readonly ElaboratedDeclaration[]): readonly string[] =>
143  declarationsOf(declarations).flatMap((d) => {
144    if (d.kind === "Rule") return [supportsRule(d)
145      ? `rule ${d.name}: satisfaction query only; no invariant enforcement`
146      : `rule ${d.name}: checked as Bool; no faithful Datalog lowering or invariant enforcement`];
147    if (d.kind === "Formula" && !supportsFormula(d)) return [`formula ${d.of}.${d.field}: no faithful Datalog lowering for this expression`];
148    return [];
149  }).concat([
150    "requires and final states: example host guards, absent from ActionTypeIR",
151    "actor type, joins and counterpart: descriptive only; no identity or authorization enforcement",
152    "Union members: checked by the prelude; stored as strings without runtime enum constraints",
153    "Many references: schema retained; membership facts must be transacted explicitly",
154  ]);

156export const encode = (field: Field, value: unknown): TripleValue => {
157  switch (valueType(field).kind) {
158    case "ref": return ref(EntityId.make(String(value)));
159    case "number": return number(Number(value));
160    case "datetime": return datetime(typeof value === "number" ? value : Date.parse(String(value)));
161    case "string": return string(String(value));
162    case "boolean": return boolean(Boolean(value));
163    default: throw new TypeError(`Unsupported seed value for ${field.name}`);
164  }
165};
166/** Stable seed ids are row-index based, e.g. room-class:2 is Deluxe. */
167export const seedOperations = (declarations: readonly ElaboratedDeclaration[]): readonly TransactOp[] => {
168  const nouns = nounsOf(declarations);
169  return declarationsOf(declarations).flatMap((d) => {
170    if (d.kind !== "Seed") return [];
171    const noun = nouns.find((n) => n.name === d.of)!;
172    return d.rows.flatMap((row, i) => Object.entries(row).flatMap(([name, value]) => {
173      const field = noun.fields.find((f) => f.name === name);
174      if (!field) throw new TypeError(`Unknown seed field ${name}`);
175      return (field.many ? value as readonly unknown[] : [value]).map((v): TransactOp => ({
176        op: "assert", entityId: EntityId.make(`${kebab(d.of)}:${i + 1}`), entityType: d.of,
177        attribute: attribute(d.of, name), value: encode(field, v),
178      }));
179    }));
180  });
181};
