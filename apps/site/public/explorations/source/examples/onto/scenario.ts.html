1import { Effect } from "effect";
2import { EntityId, KvTriples, Triples, ref, string, type TransactOp } from "@triplex-build/triplex";
3import { toTriplexConfig } from "../../src/runtime.js";
4import { elaborateHotel, lowerOnto, notLowered, seedOperations } from "./lower.js";
5import { makeStoryHost, type StepResult } from "./guards.js";
6 
7export const arrival = Date.parse("2026-11-04T15:00:00Z");
8export const reservationInput = {
9  id: "reservation:1", guest: "guest:ada", name: "Ada's stay", "room-class": "room-class:2",
10  room: "room:101", "check-in": arrival, "check-out": arrival + 3 * 86_400_000, invoice: "invoice:1",
11};
12export const serviceInput = {
13  id: "service:1", kind: "spa", charge: 75, notes: "Evening appointment", reservation: "reservation:1",
14};
15export const setupHotel = () => Effect.gen(function* () {
16  const result = elaborateHotel();
17  if (!result.ok) throw new Error(JSON.stringify(result.diagnostics));
18  const ir = lowerOnto(result.declarations);
19  const config = toTriplexConfig(ir);
20  const nodes = yield* config.nodes;
21  const triples = yield* Triples;
22  yield* triples.transact(seedOperations(result.declarations), { actor: "system:seed", commandId: "seed:room-classes" });
23  yield* triples.transact([
24    { op: "assert", entityId: EntityId.make("guest:ada"), entityType: "Guest", attribute: ":guest/name", value: string("Ada") },
25    { op: "assert", entityId: EntityId.make("room:101"), entityType: "Room", attribute: ":room/number", value: string("101") },
26    { op: "assert", entityId: EntityId.make("room:101"), entityType: "Room", attribute: ":room/room-class", value: ref(EntityId.make("room-class:2")) },
27    { op: "assert", entityId: EntityId.make("invoice:1"), entityType: "Invoice", attribute: ":invoice/reservation", value: ref(EntityId.make("reservation:1")) },
28  ], { actor: "system:fixture", commandId: "seed:hotel" });
29  return { ...makeStoryHost(result.declarations, ir), ir, notLowered: notLowered(result.declarations), configNodes: nodes.length };
30});

32export const setStatus = (owner: "Reservation" | "Service", id: string, value: string): readonly TransactOp[] => [
33  { op: "retract-pattern", pattern: { entityId: EntityId.make(id), attribute: `:${owner.toLowerCase()}/status` } },
34  { op: "assert", entityId: EntityId.make(id), entityType: owner, attribute: `:${owner.toLowerCase()}/status`, value: string(value) },

37export const playHotel = () => Effect.gen(function* () {
38  const host = yield* setupHotel();
39  const triples = yield* Triples;
40  const outcomes: StepResult[] = [];
41  outcomes.push(yield* host.invoke("hotel-booking-submits", reservationInput,
42    { actor: "guest:ada", commandId: "submit:1", now: arrival - 86_400_000 }));
43  outcomes.push(yield* host.invoke("hotel-booking-confirms", { id: "reservation:1" },
44    { actor: "system:booking", commandId: "confirm:1", now: arrival - 86_400_000 }));
45  outcomes.push(yield* host.invoke("hotel-booking-checks-in", { id: "reservation:1" },
46    { actor: "receptionist:lee", commandId: "check-in:early", now: arrival - 1 }));
47  outcomes.push(yield* host.invoke("hotel-booking-checks-in", { id: "reservation:1" },
48    { actor: "receptionist:lee", commandId: "check-in:1", now: arrival }));
49  outcomes.push(yield* host.invoke("guest-services-requests", serviceInput,
50    { actor: "guest:ada", commandId: "request:1", now: arrival }));
51  // Onto declares two independent references; it does not specify automatic inverse maintenance.
52  yield* triples.transact([
53    { op: "assert", entityId: EntityId.make("reservation:1"), entityType: "Reservation", attribute: ":reservation/services", value: ref(EntityId.make("service:1")) },
54  ], { actor: "system:fixture", commandId: "link:service" });
55  const beforeFulfilment = (yield* host.runtime.query("service-charges")).results;
56  outcomes.push(yield* host.invoke("guest-services-fulfils", { id: "service:1" },
57    { actor: "concierge:sam", commandId: "fulfil:1", now: arrival }));
58  // Reusing the creation sentence must never revive a fulfilled service.
59  outcomes.push(yield* host.invoke("guest-services-requests", serviceInput,
60    { actor: "guest:ada", commandId: "request:revive", now: arrival }));
61  const charges = (yield* host.runtime.query("service-charges")).results;
62  const dateOrder = (yield* host.runtime.query("check-out-after-check-in")).results;
63  const roomClasses = (yield* triples.queryAll({ find: ["?name", "?rate"],
64    where: [["?room", ":room-class/name", "?name"], ["?room", ":room-class/rate", "?rate"]],
65    orderBy: [{ variable: "?rate", direction: "asc" }] })).results;
66  const journal = (yield* triples.transactions()).transactions.map(({ actor, commandId }) => ({ actor, commandId }));
67  return { outcomes, queries: { roomClasses, beforeFulfilment, "service-charges": charges, "check-out-after-check-in": dateOrder }, journal, notLowered: host.notLowered };
68});

70export const runHotel = () => Effect.runPromise(playHotel().pipe(Effect.provide(KvTriples.layer)));
