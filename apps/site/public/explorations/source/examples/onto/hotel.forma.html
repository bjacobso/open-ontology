1; hotel.forma: the OntoLang hotel example, as Forma.
2 
3(project hotel "A small hotel: bookings, rooms, stays and the services guests ask for.")
4 
5; ── Actors ──
6 
7(actor Guest :person
8  "A traveler who books a room and stays at the hotel; the paying customer of the booking."
9  :joins :sign-up
10  {:name String
11   :reservations (Many Reservation)})
12 
13(actor Receptionist :person
14  "Front-desk staff who check guests in and out."
15  {:name String})
16 
17(actor Concierge :person
18  "Arranges and fulfils the services guests request during a stay."
19  {:name String})
20 
21(actor System :system
22  "The hotel booking software that confirms reservations and automates back-office steps."
23  {})
24 
25; ── Objects ──
26 
27(object RoomClass "A type of room, with its nightly rate."
28  {:name String
29   :rate Int})
30 
31(object Room "A physical room of some room class."
32  {:number String
33   :room-class (One RoomClass)})
34 
35(object Reservation "A guest's booking of a room class for a stay."
36  :final {:status [:declined :checked-out]}
37  {:guest (One Guest)
38   :name String
39   :room-class (One RoomClass)
40   :room (One Room)
41   :check-in Instant
42   :check-out Instant
43   :services (Many Service)
44   :invoice (One Invoice)
45   :status (Union :requested :confirmed :declined :checked-in :checked-out)})
46 
47(object Service
48  "A guest-requested amenity with a charge, fulfilled by the concierge; fulfilled charges roll up onto the stay."
49  :final {:status [:fulfilled]}
50  {:kind (Union :room-service :spa :laundry :taxi)
51   :charge Int
52   :notes String
53   :reservation (One Reservation)
54   :status (Union :requested :fulfilled)})
55 
56(object Invoice "What the guest owes for a stay."
57  {:reservation (One Reservation)})
58 
59; ── Rules: true whichever story runs ──
60 
61(rule check-out-after-check-in Reservation
62  "A stay ends after it starts."
63  (after? reservation.check-out reservation.check-in))
64 
65(rule past-bookings-are-settled Reservation
66  "Once the check-in date has passed, a booking is checked in, checked out or declined."
67  (or (in? reservation.status [:checked-in :checked-out :declined])
68      (not (after? now reservation.check-in))))
69 
70; ── Formulas: worked out, not typed in ──
71 
72(formula Reservation service-charges Int
73  "Charges for the services fulfilled during this stay."
74  (sum (map (fn [s] s.charge)
75            (filter (fn [s] (is? s.status :fulfilled)) reservation.services))))
76 
77(formula Invoice total Int
78  "Nights at the room-class rate, plus fulfilled services."
79  (+ (* (days-between invoice.reservation.check-in invoice.reservation.check-out)
80        invoice.reservation.room-class.rate)
81     invoice.reservation.service-charges))
82 
83; ── Stories ──
84 
85(story HotelBooking "From a guest's request to their arrival."
86  (step Guest submits Reservation :to System :new true
87    :requires [(known? reservation.room-class)]
88    :sets {:status :requested})
89  (step System confirms Reservation :to Guest
90    :requires [(is? reservation.status :requested)]
91    :sets {:status :confirmed})
92  (step Receptionist checks-in Reservation :for Guest
93    :requires [(is? reservation.status :confirmed)
94               (not (before? now reservation.check-in))]
95    :sets {:status :checked-in}))
96 
97(story GuestServices "A guest asks for something during their stay."
98  (step Guest requests Service :to Concierge :new true
99    :requires [(is? service.reservation.status :checked-in)]
100    :sets {:status :requested})
101  (step Concierge fulfils Service :for Guest
102    :requires [(is? service.status :requested)]
103    :sets {:status :fulfilled}))
104 
105; ── Seed data ──
106 
107(seed RoomClass
108  [{:name "Standard" :rate 120}
109   {:name "Deluxe" :rate 200}
110   {:name "Suite" :rate 450}])
