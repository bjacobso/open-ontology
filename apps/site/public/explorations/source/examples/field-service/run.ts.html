1import { readFile } from "node:fs/promises";
2import { Effect } from "effect";
3import { EntityId, KvTriples, Triples, string } from "@triplex-build/triplex";
4import { elaborateFormaOntology, makeOntologyRuntime, ontologyToIR } from "../../src/index.js";
5import { fieldService } from "./model.js";
6 
7const source = await readFile(new URL("./model.lisp", import.meta.url), "utf8");
8const tsIR = ontologyToIR(fieldService);
9const formaIR = elaborateFormaOntology(source, { name: "field-service", version: "1" });
10if (JSON.stringify(tsIR) !== JSON.stringify(formaIR)) {
11  throw new Error("TypeScript and Forma ontology IR differ");
12}
13 
14const runtime = makeOntologyRuntime(formaIR);
15const response = await Effect.runPromise(
16  Effect.gen(function* () {
17    const triples = yield* Triples;
18    yield* triples.transact([
19      { op: "assert", entityId: EntityId.make("technician:ada"), entityType: "Technician", attribute: ":technician/name", value: string("Ada") },
20      { op: "assert", entityId: EntityId.make("work-order:42"), entityType: "WorkOrder", attribute: ":work-order/title", value: string("Repair cooling pump") },
21      { op: "assert", entityId: EntityId.make("work-order:42"), entityType: "WorkOrder", attribute: ":work-order/status", value: string("open") },
22    ]);
23    yield* runtime.invoke("assign-work-order", { workOrder: "work-order:42", technician: "technician:ada" }, { actor: "dispatcher:grace", commandId: "assign:42", now: 1_800_000_000_000 });
24    return yield* runtime.query("assigned-work");
25  }).pipe(Effect.provide(KvTriples.layer)),
26);
27console.log(JSON.stringify(response.results, null, 2));
