1import { Schema } from "effect";
2import { Change, Datalog, Input, Ontology, Property, defineOntology } from "../../src/index.js";
3 
4const Technician = Ontology.ObjectType("Technician", {
5  properties: { name: Property.make(":technician/name", Schema.String).required() },
6});
7 
8const WorkOrder = Ontology.ObjectType("WorkOrder", {
9  properties: {
10    title: Property.make(":work-order/title", Schema.String).required(),
11    status: Property.make(":work-order/status", Schema.String).required(),
12  },
13});
14 
15const AssignedTo = Ontology.LinkType("assigned-to", {
16  from: WorkOrder,
17  to: Technician,
18  properties: { assignedAt: Property.instant(":assigned-to/assigned-at").required() },
19});
20 
21const AssignWorkOrder = Ontology.ActionType("assign-work-order", {
22  input: Schema.Struct({ workOrder: Schema.String, technician: Schema.String }),
23  changes: [
24    Change.clear(Input.value("workOrder"), WorkOrder.properties.status),
25    Change.set(Input.value("workOrder"), WorkOrder.properties.status, "assigned"),
26    Change.link(AssignedTo, Input.value("workOrder"), Input.value("technician"), {
27      ":assigned-to/assigned-at": Input.now,
28    }),
29  ],
30});
31 
32const v = Datalog.variables({
33  work: WorkOrder.entity,
34  title: Schema.String,
35  status: Schema.String,
36  assignment: AssignedTo.entity,
37  person: Technician.entity,
38  technician: Schema.String,
39  assignedAt: Schema.Date,
40});
41 
42const AssignedWork = Ontology.QueryType("assigned-work", {
43  query: Datalog.query({
44    find: [v.title, v.technician],
45    where: [
46      Datalog.match(v.work, WorkOrder.properties.title, v.title),
47      Datalog.match(v.work, WorkOrder.properties.status, v.status),
48      Datalog.predicate("!=", v.status, "cancelled"),
49      Datalog.not(Datalog.match(v.work, WorkOrder.properties.status, "archived")),
50      Datalog.match(v.assignment, AssignedTo.endpoints.from, v.work),
51      Datalog.match(v.assignment, AssignedTo.endpoints.to, v.person),
52      Datalog.match(v.assignment, AssignedTo.properties.assignedAt, v.assignedAt),
53      Datalog.match(v.person, Technician.properties.name, v.technician),
54    ],
55  }),
56});
57 
58export const fieldService = defineOntology({ name: "field-service", version: "1" }, [
59  Technician, WorkOrder, AssignedTo, AssignWorkOrder, AssignedWork,
60]);
