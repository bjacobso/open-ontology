4// A small directional model inspired by supplied schema/API context, not a production import.
5const Organization = Ontology.ObjectType("Organization", {
6  description: "The organization boundary for accounts, users, and shared configuration. Connected organizations are outside this example.",
7  properties: { name: Property.make(":organization/name", Schema.String).required() },
8});

10const Account = Ontology.ObjectType("Account", {
11  description: "An operational context inside an organization. An account is distinct from an employer; integrations and live/test modes are omitted here.",
12  properties: { organization: Property.ref(":account/organization", Organization).required() },
13});
14 
15const User = Ontology.ObjectType("User", {
16  description: "A dashboard user who may belong to organizations and groups, and be an employer assignee on a task.",
17  properties: { email: Property.make(":user/email", Schema.String).required() },
18});

20const OrganizationMembership = Ontology.ObjectType("OrganizationMembership", {

121const Task = Ontology.ObjectType("Task", {

171const StartTask = Ontology.ActionType("start-task", {
172  description: "Demo action creating an employee task against a concrete template. It does not call the supplied API or evaluate policy requirements.",
173  input: Schema.Struct({ task: Schema.String, account: Schema.String, employee: Schema.String, template: Schema.String }),
174  changes: [Change.create(Task, Input.value("task"), {
175    ":task/account": Input.value("account"),
176    ":task/employee": Input.value("employee"),
177    ":task/template": Input.value("template"),
178    ":task/status": "requires_action",
179    ":task/next-action": "employee-input",
180    ":task/started-at": Input.now,
181  })],
182});

184const RequestReview = Ontology.ActionType("request-review", {
185  description: "Demo action moving the next action to employer-review and replacing the employer assignee. It performs no permission, document, or completion checks.",
186  input: Schema.Struct({ task: Schema.String, user: Schema.String }),
187  changes: [
188    Change.clear(Input.value("task"), Task.properties.nextAction),
189    Change.set(Input.value("task"), Task.properties.nextAction, "employer-review"),
190    Change.clear(Input.value("task"), Task.properties.employerAssignee),
191    Change.set(Input.value("task"), Task.properties.employerAssignee, Input.value("user")),
192  ],
193});

202const EmployeeTasks = Ontology.QueryType("employee-tasks", {
203  description: "Joins running tasks to employees and concrete form templates. Results are not permission-filtered.",
204  query: Datalog.query({
205    find: [v.employeeName, v.formName, v.status, v.nextAction],
206    where: [
207      Datalog.match(v.task, Task.properties.employee, v.employee),
208      Datalog.match(v.task, Task.properties.template, v.template),
209      Datalog.match(v.task, Task.properties.status, v.status),
210      Datalog.match(v.task, Task.properties.nextAction, v.nextAction),
211      Datalog.match(v.employee, Employee.properties.name, v.employeeName),
212      Datalog.match(v.template, TaskTemplate.properties.name, v.formName),
213    ],
214  }),
215});

217const ReviewQueue = Ontology.QueryType("review-queue", {
218  description: "Demo query joining employee tasks needing employer review to their assigned dashboard users. It is not the production pending-employer-tasks API or an access rule.",
219  query: Datalog.query({
220    find: [v.employeeName, v.assigneeEmail],
221    where: [
222      Datalog.match(v.task, Task.properties.status, "requires_action"),
223      Datalog.match(v.task, Task.properties.nextAction, "employer-review"),
224      Datalog.match(v.task, Task.properties.employee, v.employee),
225      Datalog.match(v.employee, Employee.properties.name, v.employeeName),
226      Datalog.match(v.task, Task.properties.employerAssignee, v.user),
227      Datalog.match(v.user, User.properties.email, v.assigneeEmail),
228    ],
229  }),

232const FormVersions = Ontology.QueryType("form-versions", {
233  description: "Shows concrete templates alongside the form lineage and version status. No release or deployment operation is performed.",
234  query: Datalog.query({
235    find: [v.lineageName, v.formName, v.status],
236    where: [

240      Datalog.match(v.lineage, TaskLineage.properties.name, v.lineageName),

242    ],
243  }),
