39const distance = (a: string, b: string): number => {
40  const row = Array.from({ length: b.length + 1 }, (_, i) => i);
41  for (let i = 1; i <= a.length; i++) {
42    let previous = row[0]!;
43    row[0] = i;
44    for (let j = 1; j <= b.length; j++) {
45      const current = row[j]!;
46      row[j] = Math.min(row[j]! + 1, row[j - 1]! + 1, previous + (a[i - 1] === b[j - 1] ? 0 : 1));
47      previous = current;
48    }
49  }
50  return row[b.length]!;
51};

53const suggest = (name: string, candidates: Iterable<string>, prefix = ""): string => {
54  let best: string | undefined;
55  let bestDistance = Math.max(2, Math.floor(name.length / 3)) + 1;
56  for (const candidate of candidates) {
57    const d = distance(name, candidate);
58    if (d < bestDistance) [best, bestDistance] = [candidate, d];
59  }
60  return best === undefined ? "" : ` Did you mean ${prefix}${best}?`;

83/** Split `positional... :flag :option value` arguments, rejecting options a form does not take. */
84const options = (args: readonly SExpr[], flags: ReadonlySet<string>, valued: ReadonlySet<string>): Options => {
85  const result: Options = { positional: [], flags: new Set(), values: new Map() };
86  for (let i = 0; i < args.length; i++) {
87    const arg = args[i]!;
88    const key = keywordOf(arg);
89    if (key === undefined) {
90      result.positional.push(arg);
91      continue;
92    }
93    if (flags.has(key)) result.flags.add(key);
94    else if (valued.has(key)) {
95      if (i + 1 >= args.length) fail(arg, `:${key} needs a value`);
96      result.values.set(key, args[++i]!);
97    } else fail(arg, `Unknown option :${key}.${suggest(key, [...flags, ...valued], ":")}`);
98  }
99  return result;

183const attributeAt = (context: Context, node: SExpr, scope: Scope): Attribute => {
184  if (!isWord(node) || !node.name.includes(".")) return fail(node, `Expected an attribute path such as placement.weekly-hours, got ${show(node)}`);
185  const attribute = context.attributes.get(node.name);

187    const entity = node.name.slice(0, node.name.indexOf("."));

190      ? `Unknown attribute ${node.name}.${suggest(node.name, known)}`
191      : `Unknown entity ${entity}.${suggest(entity, [...extendable, "task"])}`);
192  }
193  const problem = scope.allows(attribute);
194  return problem ? fail(node, problem) : attribute;

197const literal = (node: SExpr, attribute: Attribute): Value => {
198  const { type } = attribute;
199  if (type.kind === "enum") {
200    const value = keywordOf(node);
201    if (value === undefined) return fail(node, `${attribute.path} is a choice; use one of ${type.options.map((o) => `:${o.value}`).join(" ")}`);
202    const values = type.options.map((option) => option.value);
203    return values.includes(value)
204      ? value
205      : fail(node, `:${value} is not an option of ${attribute.path}.${suggest(value, values, ":")}`);
206  }
207  if (type.kind === "boolean") return node._tag === "Bool" ? node.value : fail(node, `${attribute.path} is a boolean; use true or false`);
208  if (type.kind === "integer" || type.kind === "decimal") {
209    if (node._tag !== "Num") return fail(node, `${attribute.path} is ${type.kind === "integer" ? "an integer" : "a decimal"}; use a number`);
210    return type.kind === "integer" && !Number.isInteger(node.value) ? fail(node, `${attribute.path} is an integer`) : node.value;
211  }
212  if (type.kind === "date" || type.kind === "datetime") {
213    return node._tag === "Str" && /^\d{4}-\d{2}-\d{2}/.test(node.value)
214      ? node.value : fail(node, `${attribute.path} is a ${type.kind}; use an ISO date string`);
215  }
216  if (type.kind === "string") return node._tag === "Str" ? node.value : fail(node, `${attribute.path} is a string`);
217  return fail(node, `${attribute.path} is an address and cannot be compared`);

223const ruleReads = (context: Context, rule: Rule): string[] => {
224  switch (rule.op) {
225    case "and": case "or": return [...new Set(rule.args.flatMap((arg) => ruleReads(context, arg)))];
226    case "not": return ruleReads(context, rule.arg);

228    case "answer": return [];

233const elaborateRule = (context: Context, node: SExpr, scope: Scope): Rule => {
234  if (isWord(node)) {
235    if (isPath(node)) {
236      const attribute = attributeAt(context, node, scope);
237      return attribute.type.kind === "boolean"
238        ? { op: "is-true", path: attribute.path }
239        : fail(node, `${attribute.path} is a ${typeName(attribute.type)}, not a boolean; compare it with a value`);
240    }
241    const answer = scope.answers?.get(node.name);
242    if (answer) {
243      return answer.kind === "boolean" ? { op: "answer", name: node.name } : fail(node, `${node.name} is not a yes/no answer`);
244    }
245    if (context.rules.has(node.name)) {
246      for (const path of context.ruleReads.get(node.name) ?? []) {
247        const problem = scope.allows(context.attributes.get(path)!);
248        if (problem) fail(node, `${problem} (through rule ${node.name})`);
249      }
250      return { op: "rule", name: node.name };
251    }
252    const declared = context.declared.get(node.name);
253    if (context.failed.has(node.name)) throw new Cascade(node.name);
254    return fail(node, declared?.verb === "rule"
255      ? `Rule ${node.name} is used before it is defined`
256      : `Unknown rule ${node.name}.${suggest(node.name, [...context.rules.keys(), ...(scope.answers?.keys() ?? [])])}`);
257  }
258  if (node._tag !== "List" || !isSym(node.items[0])) return fail(node, `Expected a rule, got ${show(node)}`);
259  const op = node.items[0].name;
260  const args = node.items.slice(1);
261  if (op === "and" || op === "or") {
262    if (args.length === 0) fail(node, `(${op}) needs at least one rule`);
263    return { op, args: args.map((arg) => elaborateRule(context, arg, scope)) };
264  }
265  if (op === "not") {
266    if (args.length !== 1) fail(node, "(not rule) takes one rule");
267    return { op, arg: elaborateRule(context, args[0]!, scope) };
268  }
269  if (op === "missing?") {
270    if (args.length !== 1) fail(node, "(missing? path) takes one attribute");
271    return { op: "missing", path: attributeAt(context, args[0]!, scope).path };
272  }
273  if (op === "in") {
274    if (args.length !== 2) fail(node, "(in path [values]) takes an attribute and a vector");
275    const attribute = attributeAt(context, args[0]!, scope);
276    return { op, path: attribute.path, values: items(args[1], "values").map((value) => literal(value, attribute)) };
277  }
278  if (comparisons.has(op)) {
279    if (args.length !== 2) fail(node, `(${op} path value) takes an attribute and a value`);
280    const attribute = attributeAt(context, args[0]!, scope);
281    const ordered = op !== "=" && op !== "not=";
282    if (ordered && !["integer", "decimal", "date", "datetime"].includes(attribute.type.kind)) {
283      fail(args[0], `${attribute.path} is a ${typeName(attribute.type)}; ${op} needs a number or date`);
284    }
285    return { op: "compare", cmp: op as Comparison, path: attribute.path, value: literal(args[1]!, attribute) };
286  }
287  return fail(node.items[0], `Unknown rule operator ${op}.${suggest(op, ["and", "or", "not", "in", "missing?", ...comparisons])}`);
288};

290const elaborateAttributes = (context: Context, node: Extract<SExpr, { _tag: "List" }>): void => {
291  const entity = symbol(node.items[1], "an entity", node.loc);
292  if (!extendable.has(entity)) fail(node.items[1], `Cannot extend ${entity}; extend one of ${[...extendable].join(", ")}`);

297    const path = `${entity}.${attribute}`;

314      type = { kind: "enum", options: optionList };
315    } else {
316      const scalar = symbol(typeNode, "a type", list.loc);
317      if (!scalarNames.has(scalar)) fail(typeNode, `Unknown type ${scalar}.${suggest(scalar, [...scalarNames, "one-of"])}`);
318      type = { kind: scalar as Scalar };
319    }
320    const label = string(list.items[2], `a label for ${path}`, list.loc).trim();
321    if (!label) fail(list.items[2], "Labels must not be empty");
322    const opts = options(list.items.slice(3), new Set(["searchable", "sensitive", "scope"]), new Set(["pattern", "deprecated"]));
323    if (opts.positional.length) fail(opts.positional[0], `Unexpected ${show(opts.positional[0]!)}`);
324    const flagged = (flag: string) => opts.flags.has(flag);
325    if (flagged("sensitive") && (flagged("searchable") || flagged("scope"))) {
326      fail(list, `${path} is :sensitive, so it cannot be :searchable or :scope`);

328    if (type.kind === "address" && opts.flags.size > 0) fail(list, "Address attributes cannot be :searchable, :sensitive, or :scope");

339    context.attributes.set(path, {

342      ...(pattern ? { pattern: string(pattern, "a pattern") } : {}),

368/** Build-time arithmetic folds to a number; runtime formulas keep attribute and answer references. */
369const elaborateFormula = (context: Context, node: SExpr, scope: Scope): Formula => {
370  if (node._tag === "Num") return { op: "number", value: node.value };
371  if (isPath(node)) {
372    const attribute = attributeAt(context, node, scope);
373    return attribute.type.kind === "integer" || attribute.type.kind === "decimal"
374      ? { op: "path", path: attribute.path } : fail(node, `${attribute.path} is not a number`);
375  }
376  if (isWord(node)) {
377    const constant = context.constants.get(node.name);
378    if (constant !== undefined) return { op: "number", value: constant };
379    const answer = scope.answers?.get(node.name);
380    if (answer?.kind === "integer" || answer?.kind === "decimal") return { op: "answer", name: node.name };
381    return fail(node, `Unknown number ${node.name}.${suggest(node.name, context.constants.keys())}`);
382  }
383  const fn = headOf(node);
384  if (!fn || !formulaFns.has(fn)) return fail(node, `Formulas use numbers, attributes, ${[...formulaFns].join(" ")}`);
385  const args = (node as Extract<SExpr, { _tag: "List" }>).items.slice(1).map((arg) => elaborateFormula(context, arg, scope));
386  if (fn === "round" && args.length !== 2) fail(node, "(round value places) takes two arguments");
387  if (args.length < 1) fail(node, `(${fn}) needs arguments`);
388  return { op: "call", fn: fn as "round", args };
389};

391const foldFormula = (formula: Formula): number | undefined => {
392  if (formula.op === "number") return formula.value;
393  if (formula.op !== "call") return undefined;
394  const values = formula.args.map(foldFormula);
395  if (values.some((value) => value === undefined)) return undefined;
396  const [first, ...rest] = values as number[];
397  switch (formula.fn) {
398    case "+": return rest.reduce((a, b) => a + b, first!);
399    case "-": return rest.length ? rest.reduce((a, b) => a - b, first!) : -first!;
400    case "*": return rest.reduce((a, b) => a * b, first!);
401    case "/": return rest.reduce((a, b) => a / b, first!);
402    case "round": return Number(first!.toFixed(rest[0]));
403  }

418const elaborateForm = (context: Context, node: Extract<SExpr, { _tag: "List" }>): Form => {
419  const account = context.account!;
420  const formName = name(node.items[1], "a form name", node.loc);
421  const hasTitle = node.items[2]?._tag === "Str";
422  const rest = node.items.slice(hasTitle ? 3 : 2);
423  const opts = options(rest, new Set(), new Set(["key", "due-in", "from"]));
424  const keyNode = opts.values.get("key");
425  const key = keyNode ? string(keyNode, "a form key") : `${account.name}.${formName}`;
426  const fromNode = opts.values.get("from");
427  if (fromNode) {
428    if (!keyNode) fail(node, "A form :from another package needs its :key");
429    if (key.startsWith(`${account.name}.`)) fail(keyNode, `${key} is in this account's namespace; a :from form belongs to another package`);
430    if (hasTitle || opts.positional.length || opts.values.has("due-in")) fail(node, "A :from form is a reference; it has no title, pages, or :due-in");
431    return { name: formName, key, from: string(fromNode, "a package name"), pages: [], reads: [], writes: [], loc: node.loc };
432  }
433  if (!hasTitle) fail(node, `(form ${formName} "Title" pages...) needs a title`, node.loc);
434  if (keyNode && !key.startsWith(`${account.name}.`)) fail(keyNode, `Form keys must start with ${account.name}.`);
435  const dueNode = opts.values.get("due-in");
436  const dueIn = dueNode ? number(dueNode, "days") : undefined;
437  if (dueIn !== undefined && (!Number.isInteger(dueIn) || dueIn <= 0)) fail(dueNode, ":due-in is a positive number of days");
438 
439  const answers = new Map<string, AttributeType>();
440  const scope: Scope = { ...placementScope("Forms"), answers };
441  const reads: string[] = [];
442  const writes: string[] = [];
443  const read = (paths: readonly string[]) => reads.push(...paths.filter((path) => !reads.includes(path)));
444  const translationKeys = new Set(account.languages.slice(1));
445  const defineAnswer = (node: SExpr, type: AttributeType): FieldTarget => {
446    const answer = name(node, "an answer name");
447    if (answers.has(answer)) fail(node, `${answer} is already an answer in ${formName}`);
448    answers.set(answer, type);
449    return { answer };
450  };
451  const targetRef = (node: SExpr): FieldTarget => {
452    if (isPath(node)) {
453      const path = attributeAt(context, node, scope).path;
454      read([path]);
455      return { path };
456    }
457    const answer = symbol(node, "an attribute or answer");
458    return answers.has(answer) ? { answer } : fail(node, `Unknown answer ${answer}.${suggest(answer, answers.keys())}`);
459  };
460 
461  const pages: Page[] = opts.positional.map((pageNode): Page => {
462    if (headOf(pageNode) !== "page") return fail(pageNode, `Expected (page "Title" :employee fields...)`);
463    const pageItems = (pageNode as Extract<SExpr, { _tag: "List" }>).items;
464    const title = string(pageItems[1], "a page title", pageNode.loc);
465    const assigneeNode = pageItems[2];
466    const assignee = keywordOf(assigneeNode) ?? "";
467    if (assignee !== "employee" && assignee !== "employer") fail(assigneeNode ?? pageNode, "Pages are assigned to :employee or :employer");
468    const fields = pageItems.slice(3).map((fieldNode): Field => {
469      const kind = headOf(fieldNode);
470      const args = fieldNode._tag === "List" ? fieldNode.items.slice(1) : [];
471      if (kind === "signature") {
472        return { kind, target: defineAnswer(args[0]!, { kind: "string" }), label: "Signature", required: true, translations: {} };
473      }
474      if (kind === "content") return { kind, text: string(args[0], "content text", fieldNode.loc), required: false, translations: {} };
475      if (kind === "review") {
476        return { kind, label: "Review", rows: items(args[0], "review rows", fieldNode.loc).map(targetRef), required: false, translations: {} };
477      }
478      if (kind === "formula") {
479        const label = string(args[1], "a formula label", fieldNode.loc);
480        const formula = elaborateFormula(context, args[2] ?? fail(fieldNode, "(formula name \"Label\" expression)"), scope);
481        read(formulaReads(formula));
482        return { kind, target: defineAnswer(args[0]!, { kind: "decimal" }), label, formula, required: false, translations: {} };
483      }
484      const fieldType = kind ? fieldTypes[kind] : undefined;
485      if (!fieldType) {
486        return fail(fieldNode, `Unknown field ${kind ?? show(fieldNode)}.${suggest(kind ?? "", [...Object.keys(fieldTypes), "signature", "formula", "review", "content"])}`);
487      }
488      const targetNode = args[0] ?? fail(fieldNode, `(${kind} target "Label") needs a target`);
489      const label = string(args[1], "a field label", fieldNode.loc);
490      const fieldOpts = options(args.slice(2), new Set(["required"]), new Set(["show-if", ...translationKeys]));
491      if (fieldOpts.positional.length) fail(fieldOpts.positional[0], `Unexpected ${show(fieldOpts.positional[0]!)}`);
492      let target: FieldTarget;
493      if (isPath(targetNode)) {
494        const attribute = attributeAt(context, targetNode, scope);
495        if (!fieldType.accepts(attribute.type)) {
496          fail(targetNode, `A ${kind} field writes ${fieldType.expects} values; ${attribute.path} is a ${typeName(attribute.type)}`);
497        }
498        if (attribute.deprecated) fail(targetNode, `${attribute.path} is deprecated (${attribute.deprecated}); forms cannot write it`);
499        if (attribute.builtin) fail(targetNode, `${attribute.path} is a built-in attribute; forms write custom attributes`);
500        if (writes.includes(attribute.path)) fail(targetNode, `${formName} already writes ${attribute.path}`);
501        writes.push(attribute.path);
502        target = { path: attribute.path };
503      } else {
504        if (!fieldType.answer) fail(targetNode, `A ${kind} field must bind a choice attribute`);
505        target = defineAnswer(targetNode, fieldType.answer!);
506      }
507      const showIfNode = fieldOpts.values.get("show-if");
508      const showIf = showIfNode ? elaborateRule(context, showIfNode, scope) : undefined;
509      if (showIf) read(ruleReads(context, showIf));
510      const translations = Object.fromEntries([...translationKeys].filter((language) => fieldOpts.values.has(language))
511        .map((language) => [language, string(fieldOpts.values.get(language), `the ${language} label`)]));
512      return { kind: kind!, target, label, required: fieldOpts.flags.has("required"), ...(showIf ? { showIf } : {}), translations };
513    });
514    return { title, assignee: assignee as Page["assignee"], fields };
515  });
516  if (pages.length === 0) fail(node, `${formName} needs at least one page`);
517  return {
518    name: formName, key, title: string(node.items[2], "a title"), ...(dueIn !== undefined ? { dueIn } : {}),
519    pages, reads: reads.filter((path) => !writes.includes(path)), writes, loc: node.loc,
520  };
521};

547const elaborateAutomation = (context: Context, node: Extract<SExpr, { _tag: "List" }>): Automation => {
548  const automationName = name(node.items[1], "an automation name", node.loc);
549  const title = string(node.items[2], "an automation title", node.loc);
550  const opts = options(node.items.slice(3), new Set(), new Set(["on"]));
551  const triggerNode = opts.values.get("on") ?? fail(node, `${automationName} needs a trigger, such as :on (task-created form)`);
552  const triggerKind = headOf(triggerNode);
553  let trigger: Trigger;
554  if (triggerKind === "task-created" || triggerKind === "task-completed") {
555    trigger = { kind: triggerKind, form: declaredAs(context, (triggerNode as Extract<SExpr, { _tag: "List" }>).items[1] ?? triggerNode, "form") };
556  } else if (triggerKind === "placement-created") {
557    trigger = { kind: triggerKind };
558  } else {
559    return fail(triggerNode, `Unknown trigger ${triggerKind ?? show(triggerNode)}.${suggest(triggerKind ?? "", ["task-created", "task-completed", "placement-created"])}`);
560  }
561  const scope = trigger.kind === "placement-created" ? placementScope("Placement triggers") : taskScope;
562  const counts = new Map<string, number>();
563  const id = (kind: string) => {
564    const count = (counts.get(kind) ?? 0) + 1;
565    counts.set(kind, count);
566    return count === 1 ? kind : `${kind}-${count}`;
567  };
568  const steps = (nodes: readonly SExpr[]): Step[] => nodes.map((stepNode): Step => {
569    const kind = headOf(stepNode);
570    const args = stepNode._tag === "List" ? stepNode.items.slice(1) : [];
571    if (kind === "wait") {
572      const days = number(args[0], "a number of days", stepNode.loc);
573      const waitOpts = options(args.slice(1), new Set(["days"]), new Set(["after"]));
574      if (!waitOpts.flags.has("days")) fail(stepNode, "(wait n :days) names its unit");
575      const afterNode = waitOpts.values.get("after");
576      if (!afterNode) return { kind, id: id(kind), days };
577      const after = attributeAt(context, afterNode, scope);
578      if (after.type.kind !== "date" && after.type.kind !== "datetime") fail(afterNode, `${after.path} is not a date`);
579      if (after.path === "task.due-at" && trigger.kind !== "placement-created") {
580        const form = context.forms.get(trigger.form);
581        if (form && !form.from && form.dueIn === undefined) {
582          fail(afterNode, `${automationName} waits after task.due-at, but ${form.name} sets no :due-in`);
583        }
584      }
585      return { kind, id: id(kind), days, after: after.path };
586    }
587    if (kind === "when") {
588      const stepId = id(kind);
589      return { kind, id: stepId, rule: elaborateRule(context, args[0] ?? fail(stepNode, "(when rule steps...)"), scope), then: steps(args.slice(1)) };
590    }
591    if (kind === "email") {
592      const to = keywordOf(args[0]) ?? "";
593      if (to !== "employee" && to !== "employer") fail(args[0] ?? stepNode, "Email :employee or :employer");
594      const template = declaredAs(context, args[1] ?? stepNode, "param");
595      const param = context.params.get(template);
596      if (param && param.type !== "sendgrid-template") fail(args[1], `${template} is a ${param.type}, not a :sendgrid-template`);
597      const subject = string(args[2], "an email subject", stepNode.loc);
598      const body = string(args[3], "an email body", stepNode.loc);
599      for (const match of body.matchAll(/\{\{\s*([a-z_]+)\.([a-z_]+)\s*\}\}/g)) {
600        const path = `${match[1]}.${match[2]!.replaceAll("_", "-")}`;
601        const attribute = context.attributes.get(path);
602        if (!attribute) fail(args[3], `The body renders unknown attribute ${match[1]}.${match[2]}`);
603        const problem = scope.allows(attribute!);
604        if (problem) fail(args[3], problem);
605        if (attribute!.sensitive) {
606          context.diagnostics.push({ severity: "warning", message: `The email renders sensitive attribute ${path}`, loc: args[3]!.loc });
607        }
608      }
609      return { kind, id: id(kind), to: to as "employee", template, subject, body };
610    }
611    if (kind === "assign") return { kind, id: id(kind), form: declaredAs(context, args[0] ?? stepNode, "form") };
612    return fail(stepNode, `Unknown step ${kind ?? show(stepNode)}.${suggest(kind ?? "", ["wait", "when", "email", "assign"])}`);
613  });
614  const body = steps(opts.positional);
615  if (body.length === 0) fail(node, `${automationName} has no steps`);
616  return { name: automationName, key: `${context.account!.name}.${automationName}`, title, trigger, steps: body, loc: node.loc };

668const elaborateScenario = (context: Context, node: Extract<SExpr, { _tag: "List" }>): Scenario => {
669  const title = string(node.items[1], "a scenario title", node.loc);
670  const givenNode = node.items[2];
671  if (givenNode?._tag !== "Map") return fail(givenNode ?? node, "A scenario starts with its facts: {path value ...}");
672  const factScope: Scope = {
673    allows: (attribute) => attribute.entity === "task" ? "Scenarios give placement facts; task facts come from the simulation" : undefined,
674  };
675  const facts = (pairs: readonly (readonly [SExpr, SExpr])[]): Map<string, Value> =>
676    new Map(pairs.map(([key, value]) => {
677      const attribute = attributeAt(context, key, factScope);
678      return [attribute.path, literal(value, attribute)];
679    }));
680  const given = facts(givenNode.pairs);
681  const names = (node: SExpr | undefined, verb: string): string[] =>
682    items(node, `${verb} names`).map((item) => declaredAs(context, item, verb));
683  const expectation = (args: readonly SExpr[]): Expectation => {
684    const opts = options(args, new Set(), new Set(["policies", "forms", "unknown", "tasks", "open", "emails"]));
685    if (opts.positional.length) fail(opts.positional[0], `Unexpected ${show(opts.positional[0]!)}`);
686    const expect: Record<string, unknown> = {};
687    if (opts.values.has("policies")) expect.policies = names(opts.values.get("policies"), "policy");
688    for (const key of ["forms", "tasks", "open"]) if (opts.values.has(key)) expect[key] = names(opts.values.get(key), "form");
689    const unknown = opts.values.get("unknown");
690    if (unknown) {
691      if (unknown._tag !== "Map") fail(unknown, ":unknown maps a policy or form to the facts it waits for");
692      expect.unknown = Object.fromEntries((unknown as Extract<SExpr, { _tag: "Map" }>).pairs.map(([key, paths]) => {
693        const subject = symbol(key, "a policy or form");
694        const verb = context.declared.get(subject)?.verb;
695        if (verb !== "policy" && verb !== "form") fail(key, `Unknown policy or form ${subject}`);
696        return [subject, items(paths, "attribute paths").map((path) => attributeAt(context, path, anyScope).path)];
697      }));
698    }
699    const emails = opts.values.get("emails");
700    if (emails) {
701      expect.emails = items(emails, "emails").map((email): EmailExpectation => {
702        if (email._tag === "Str") return { subject: email.value };
703        if (email._tag !== "Map") return fail(email, "An expected email is a subject or {:to :subject :body}");
704        return Object.fromEntries(email.pairs.map(([key, value]) => {
705          const field = keywordOf(key) ?? "";
706          if (!["to", "subject", "body"].includes(field)) fail(key, "Expected email keys are :to, :subject, and :body");
707          return [field, field === "to" ? settingValue(value) : string(value, field)];
708        }));
709      });
710    }
711    return expect as Expectation;
712  };
713  const action = (actionNode: SExpr): ScenarioAction => {
714    const kind = headOf(actionNode);
715    const args = actionNode._tag === "List" ? actionNode.items.slice(1) : [];
716    if (kind === "expect") return { kind, expect: expectation(args), loc: actionNode.loc };
717    if (kind === "record") {
718      const attribute = attributeAt(context, args[0] ?? actionNode, factScope);
719      return { kind, path: attribute.path, value: literal(args[1] ?? fail(actionNode, "(record path value)"), attribute), loc: actionNode.loc };
720    }
721    if (kind === "complete") {
722      const form = declaredAs(context, args[0] ?? actionNode, "form");
723      const writesNode = args[1];
724      if (writesNode && writesNode._tag !== "Map") fail(writesNode, "(complete form {path value ...})");
725      const writes = writesNode ? facts((writesNode as Extract<SExpr, { _tag: "Map" }>).pairs) : new Map<string, Value>();
726      const declared = context.forms.get(form)?.writes ?? [];
727      for (const [path] of writes) if (!declared.includes(path)) fail(writesNode, `${form} does not write ${path}`);
728      return { kind, form, writes, loc: actionNode.loc };
729    }
730    return fail(actionNode, `Unknown scenario step ${kind ?? show(actionNode)}.${suggest(kind ?? "", ["expect", "record", "complete", "day"])}`);
731  };
732  const days: Array<{ day: number; actions: ScenarioAction[] }> = [{ day: 0, actions: [] }];
733  for (const step of node.items.slice(3)) {
734    if (headOf(step) === "day") {
735      const dayItems = (step as Extract<SExpr, { _tag: "List" }>).items;
736      const day = number(dayItems[1], "a day number", step.loc);
737      if (!Number.isInteger(day) || day <= days[days.length - 1]!.day) fail(dayItems[1], "Scenario days must be whole numbers that increase");
738      days.push({ day, actions: dayItems.slice(2).map(action) });
739    } else {
740      if (days.length > 1) fail(step, "Steps after a (day n ...) belong inside a day");
741      days[0]!.actions.push(action(step));
742    }
743  }
744  return { title, given, days, loc: node.loc };
745};

763/** Elaborate one account source. Every problem becomes a located diagnostic; there are no partial programs. */
764export const elaborateAccount = (source: string): ElaborationResult => {
765  let expressions: readonly SExpr[];
766  try {
767    const parsed = parse(source);
768    if (parsed.errors.length) {
769      const error = parsed.errors[0]!;
770      return { diagnostics: [{ severity: "error", message: error.message, ...(error.loc ? { loc: error.loc } : {}) }] };
771    }
772    expressions = toSExprMany(parsed.redTree);
773  } catch (error) {
774    const loc = (error as { loc?: Loc }).loc;
775    return { diagnostics: [{ severity: "error", message: (error as Error).message, ...(loc ? { loc } : {}) }] };
776  }
777 
778  const context: Context = {
779    attributes: new Map(builtinAttributes.map((attribute) => [attribute.path, attribute])),
780    rules: new Map(), ruleReads: new Map(), constants: new Map(), declared: new Map(), failed: new Set(),
781    forms: new Map(), params: new Map(), diagnostics: [],
782  };
783  const report = (error: unknown) => {
784    if (!(error instanceof AccountError)) throw error;
785    if (error instanceof Cascade) return;
786    context.diagnostics.push({ severity: "error", message: error.message, ...(error.loc ? { loc: error.loc } : {}) });
787  };
788 
789  const declarations: Declaration[] = [];
790  const lists: Array<Extract<SExpr, { _tag: "List" }>> = [];
791  for (const expression of expressions) {
792    try {
793      const verb = headOf(expression);
794      if (!verb || !verbs.has(verb)) {
795        fail(expression._tag === "List" ? expression.items[0] ?? expression : expression,
796          `Unknown declaration ${verb ?? show(expression)}.${suggest(verb ?? "", verbs)}`);
797      }
798      const list = expression as Extract<SExpr, { _tag: "List" }>;
799      const label = verb === "scenario" ? string(list.items[1], "a scenario title", list.loc)
800        : verb === "attributes" ? symbol(list.items[1], "an entity", list.loc)
801        : name(list.items[1], `a ${verb} name`, list.loc);
802      if (namedVerbs.has(verb!)) {
803        const previous = context.declared.get(label);
804        if (previous) fail(list.items[1], `${label} is already declared on line ${previous.loc.line}`);
805        context.declared.set(label, { verb: verb!, loc: list.loc });
806      }
807      declarations.push({ verb: verb!, name: label, startLine: list.loc.line, endLine: lineAt(source, list.loc.end - 1) });
808      lists.push(list);
809    } catch (error) {
810      report(error);
811    }
812  }
813  const accounts = lists.filter((list) => headOf(list) === "account");
814  if (accounts.length !== 1 || lists[0] !== accounts[0]) {
815    context.diagnostics.push({
816      severity: "error", message: "An account file starts with exactly one (account name \"Title\" ...)",
817      loc: (accounts[1] ?? lists[0])?.loc ?? { line: 1, col: 1, start: 0, end: 0 },
818    });
819    return { diagnostics: context.diagnostics };
820  }
821 
822  const policies: Policy[] = [];
823  const automations: Automation[] = [];
824  const groups: Group[] = [];
825  const integrations: Integration[] = [];
826  const scenarios: Scenario[] = [];
827  for (const current of order) {
828    for (const list of lists.filter((item) => phase(headOf(item)!) === current)) {
829      try {
830        const verb = headOf(list)!;
831        if (verb === "account") context.account = elaborateHeader(context, list);
832        else if (verb === "attributes") elaborateAttributes(context, list);
833        else if (verb === "def") {
834          const constant = foldFormula(elaborateFormula(context, list.items[2] ?? fail(list, "(def name expression)"), anyScope));
835          if (constant === undefined) fail(list.items[2], "A def is evaluated at build time; use numbers and other defs");
836          context.constants.set(symbol(list.items[1], "a name"), constant!);
837        } else if (verb === "rule") {
838          const ruleName = symbol(list.items[1], "a rule name");
839          if (list.items.length !== 3) fail(list, "(rule name expression)");
840          const rule = elaborateRule(context, list.items[2]!, anyScope);
841          context.rules.set(ruleName, rule);
842          context.ruleReads.set(ruleName, ruleReads(context, rule));
843        } else if (verb === "form") {
844          const form = elaborateForm(context, list);
845          context.forms.set(form.name, form);
846        } else if (verb === "param") {
847          const param = elaborateParam(context, list);
848          context.params.set(param.name, param);
849        } else if (verb === "policy") policies.push(elaboratePolicy(context, list));
850        else if (verb === "automation") automations.push(elaborateAutomation(context, list));
851        else if (verb === "group") groups.push(elaborateGroup(context, list));
852        else if (verb === "integration") integrations.push(elaborateIntegration(context, list));
853        else scenarios.push(elaborateScenario(context, list));
854      } catch (error) {
855        if (isWord(list.items[1])) context.failed.add(list.items[1].name);
856        report(error);
857      }
858      if (current === "account" && !context.account) return { diagnostics: context.diagnostics };
859    }
860  }
861  context.diagnostics.sort((a, b) => (a.loc?.start ?? 0) - (b.loc?.start ?? 0));
862  if (context.diagnostics.some((diagnostic) => diagnostic.severity === "error")) return { diagnostics: context.diagnostics };
863  const forms = new Set([...policies.flatMap((policy) => policy.require.map((entry) => entry.form)),
864    ...automations.flatMap((automation) => {
865      const assigned = (steps: readonly Step[]): string[] => steps.flatMap((step) =>
866        step.kind === "assign" ? [step.form] : step.kind === "when" ? assigned(step.then) : []);
867      return [...assigned(automation.steps), ...("form" in automation.trigger ? [automation.trigger.form] : [])];
868    })]);
869  for (const form of context.forms.values()) {
870    if (!forms.has(form.name)) {
871      context.diagnostics.push({ severity: "warning", message: `No policy or automation uses ${form.name}`, loc: form.loc });
872    }
873  }
874  context.diagnostics.sort((a, b) => (a.loc?.start ?? 0) - (b.loc?.start ?? 0));
875  return {
876    program: {
877      account: context.account!, attributes: context.attributes, rules: context.rules, constants: context.constants,
878      forms: context.forms, policies, params: context.params, automations, groups, integrations, scenarios, declarations,
879    },
880    diagnostics: context.diagnostics,
881  };
882};
