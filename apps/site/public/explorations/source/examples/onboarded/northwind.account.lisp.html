1; Northwind: one Onboarded account, end to end, in one file.
2;
3; Illustrative and synthetic. The Forma reader parses this file; the example's account
4; elaborator checks it, lowers it to directional resource IR, and simulates its scenarios.
5; Nothing here calls Onboarded or deploys configuration.
6;
7;   pnpm account:check examples/onboarded/northwind.account.lisp
8 
9(account northwind "Northwind Logistics"
10  :languages [:en :es]
11  :sender "onboarding@northwind.example"
12  :support "people@northwind.example"
13  :dedupe [employee.email])
14 
15; ── Attributes: built-in entities, extended with typed fields ────────────────────────────
16 
17(attributes client
18  (region (one-of :west "West" :east "East") "Region" :scope)
19  (requires-drug-screen boolean "Requires drug screen"))
20 
21(attributes job
22  (site-type (one-of :warehouse "Warehouse" :office "Office") "Site type" :searchable)
23  (hourly-rate decimal "Hourly rate"))
24 
25(attributes placement
26  (union-local (one-of :local-42 "Local 42" :local-77 "Local 77" :none "Not represented")
27    "Union local" :searchable)
28  (weekly-hours integer "Scheduled weekly hours"))
29 
30(attributes employee
31  (badge-number string "Badge number" :searchable :pattern "^NW-[0-9]{6}$")
32  (union-card-number string "Union card number" :sensitive)
33  (mailing-address address "Mailing address")
34  (forklift-certified-on date "Forklift certification date")
35  (legacy-hris-id string "Legacy HRIS ID" :deprecated "Replaced by badge-number"))
36 
37; ── Rules: named, three-valued, usable by policies, forms, automations, and access ───────
38 
39(rule at-warehouse (= job.site-type :warehouse))
40 
41(rule union-covered
42  (and (not= placement.union-local :none)
43       (>= placement.weekly-hours 20)))
44 
45(def dues-rate (/ 1.5 100))   ; build-time arithmetic, inlined as 0.015
46 
47; ── Forms: fields bind attributes or form answers; reads and writes are inferred ─────────
48 
49(form w4 :from "onboarded.compliance" :key "irs.w-4")
50 
51(form union-card "Union card"
52  :due-in 30
53  (page "Union membership" :employee
54    (checkbox has-card "I already hold a union card" :es "Ya tengo una tarjeta sindical")
55    (text employee.union-card-number "Union card number" :required :show-if has-card)
56    (address employee.mailing-address "Mailing address" :required)
57    (formula weekly-dues "Estimated weekly dues"
58      (round (* job.hourly-rate placement.weekly-hours dues-rate) 2))
59    (signature employee-signature))
60  (page "Steward review" :employer
61    (review [employee.union-card-number weekly-dues])
62    (signature steward-signature)))
63 
64(form site-safety "Globex site safety" :key "northwind.globex.site-safety"
65  (page "Site rules" :employee
66    (checkbox acknowledged "I have read the Globex site safety rules" :required)
67    (signature employee-signature)))
68 
69(form forklift-certification "Forklift certification"
70  (page "Certification" :employee
71    (date employee.forklift-certified-on "Certification date" :required)))
72 
73(form drug-screen-consent "Drug screen consent"
74  (page "Consent" :employee
75    (content "I consent to a pre-placement drug screen.")
76    (signature consent-signature)))
77 
78; ── Policies: when a placement matches, require forms ────────────────────────────────────
79 
80(policy warehouse-onboarding "Warehouse onboarding"
81  :when at-warehouse
82  :require [w4 site-safety
83            (drug-screen-consent :when client.requires-drug-screen)])
84 
85(policy union-card-obligation "Union card"
86  :when union-covered
87  :require [union-card])
88 
89; ── Automations: a trigger, then steps ───────────────────────────────────────────────────
90 
91(param overdue-template :sendgrid-template "Supplied per target account.")
92 
93(automation union-card-reminder "Union card reminder"
94  :on (task-created union-card)
95  (wait 2 :days :after task.due-at)
96  (when (not= task.status :completed)
97    (email :employee overdue-template "Your union card is overdue"
98      "Hi {{ employee.first_name }}, please complete your union card.")))
99 
100(automation forklift-follow-up "Request forklift certification"
101  :on (placement-created)
102  (when (and at-warehouse (missing? employee.forklift-certified-on))
103    (assign forklift-certification)))
104 
105; ── Access and integrations ──────────────────────────────────────────────────────────────
106 
107(group west-managers "West region managers"
108  :can [:placement/read :employee/read :task/read :task/edit]
109  :sees (= client.region :west))
110 
111(integration checkr "@onboarded-platform/checkr@1.4.0"
112  :connection "checkr"
113  :package "driver_pro"
114  :adjudication :manual)

116; ── Scenarios: facts in, outcomes out; every decision covered both ways ──────────────────
117 
118(scenario "warehouse, Local 42, full time"
119  {job.site-type :warehouse placement.union-local :local-42
120   placement.weekly-hours 32 client.requires-drug-screen false}
121  (expect :policies [warehouse-onboarding union-card-obligation]
122          :forms [w4 site-safety union-card]
123          :tasks [w4 site-safety union-card forklift-certification]))
124 
125(scenario "certified driver, drug screen required"
126  {job.site-type :warehouse placement.union-local :none
127   client.requires-drug-screen true employee.forklift-certified-on "2026-03-14"}
128  (expect :policies [warehouse-onboarding]
129          :forms [w4 site-safety drug-screen-consent]
130          :tasks [w4 site-safety drug-screen-consent]))
131 
132(scenario "office, not represented"
133  {job.site-type :office placement.union-local :none}
134  (expect :policies [] :forms [] :tasks []))
135 
136(scenario "union local recorded on day five"
137  {job.site-type :office placement.weekly-hours 32}
138  (expect :forms [] :unknown {union-card-obligation [placement.union-local]})
139  (day 5 (record placement.union-local :local-42)
140         (expect :policies [union-card-obligation] :tasks [union-card])))
141 
142(scenario "Ada returns her union card on day ten"
143  {employee.first-name "Ada" job.site-type :office
144   placement.union-local :local-42 placement.weekly-hours 32}
145  (day 10 (complete union-card))
146  (day 40 (expect :open [] :emails [])))
147 
148(scenario "Ben's union card goes overdue"
149  {employee.first-name "Ben" job.site-type :office
150   placement.union-local :local-77 placement.weekly-hours 24}
151  (day 31 (expect :emails []))
152  (day 32 (expect :open [union-card]
153                  :emails [{:to :employee :body "Hi Ben, please complete your union card."}])))
