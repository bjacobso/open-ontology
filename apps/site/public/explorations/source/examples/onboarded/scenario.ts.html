1import { Effect } from "effect";
2import { EntityId, KvTriples, Triples, boolean, ref, string } from "@triplex-build/triplex";

4import { onboarded } from "./model.js";

6/** Synthetic custom-property definitions, including permission-scope flags. */
7export const permissionProperties = [
8  { path: "employee.department", scalarType: "string", isPermissionScope: true },
9  { path: "employee.work_location", scalarType: "string", isPermissionScope: true },
10  { path: "employee.start_date", scalarType: "date", isPermissionScope: false },
11] as const;

13type SeedValue = ReturnType<typeof string> | ReturnType<typeof ref> | ReturnType<typeof boolean>;
14const reference = (id: string) => ref(EntityId.make(id));
15const record = (entityType: string, id: string, properties: Readonly<Record<string, SeedValue>>) =>
16  Object.entries(properties).map(([attribute, value]) => ({
17    op: "assert" as const, entityId: EntityId.make(id), entityType, attribute, value,
18  }));

20/** Synthetic data only: no API calls, production config, personal documents, or persistence. */
21export const runOnboardedScenario = () => {
