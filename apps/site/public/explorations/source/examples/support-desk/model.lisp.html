1(define-entity Customer
2  (:doc "An organisation that raises tickets.")
3  (:field [customer/name String {:required true}]))
4 
5(define-entity Engineer
6  (:field [engineer/name String {:required true}]))
7 
8(define-entity Ticket
9  (:field [ticket/title String {:required true}])
10  (:field [ticket/customer Customer {:required true}])
11  (:field [ticket/status String {:required true}])
12  (:field [ticket/priority Int {:required true}])
13  (:field [ticket/opened-at Instant {:required true}]))
14 
15(define-relation escalated-to Ticket Engineer
16  (:field [escalated-to/reason String {:required true}])
17  (:field [escalated-to/at Instant {:required true}]))
18 
19(define-action open-ticket
20  (:input [ticket String {:required true}])
21  (:input [customer String {:required true}])
22  (:input [title String {:required true}])
23  (:input [priority Int {:required true}])
24  (:returns String)
25  (:do (create Ticket ticket
26         {:ticket/title title
27          :ticket/customer customer
28          :ticket/status "open"
29          :ticket/priority priority
30          :ticket/opened-at now})))
31 
32(define-action escalate-ticket
33  (:input [ticket String {:required true}])
34  (:input [engineer String {:required true}])
35  (:input [reason String {:required true}])
36  (:returns String)
37  (:do (changes
38    (clear ticket :ticket/status)
39    (set ticket :ticket/status "escalated")
40    (link escalated-to ticket engineer
41      {:escalated-to/reason reason :escalated-to/at now}))))
42 
43(define-action close-ticket
44  (:input [ticket String {:required true}])
45  (:returns String)
46  (:do (changes
47    (clear ticket :ticket/status)
48    (set ticket :ticket/status "closed"))))

50(define-datalog-query open-work
51  (:query
52    {:find ["?title" "?customer" "?priority"]
53     :where [["?t" ":ticket/title" "?title"]
54             ["?t" ":ticket/priority" "?priority"]
55             ["?t" ":ticket/customer" "?c"]
56             ["?c" ":customer/name" "?customer"]
57             ["not" ["?t" ":ticket/status" "closed"]]]
58     :orderBy [{:variable "?priority" :direction "desc"}]}))

60(define-datalog-query escalations
