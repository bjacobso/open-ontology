1import { readFile } from "node:fs/promises";
2import { Effect } from "effect";
3import { EntityId, KvTriples, Triples, string } from "@triplex-build/triplex";
4import { elaborateFormaOntology, makeOntologyRuntime } from "../../src/index.js";
5 
6const source = await readFile(new URL("./model.lisp", import.meta.url), "utf8");
7const runtime = makeOntologyRuntime(elaborateFormaOntology(source, { name: "support-desk", version: "1" }));
8 
9const results = await Effect.runPromise(
10  Effect.gen(function* () {
11    const triples = yield* Triples;
12    // Seed the facts that exist before any action runs.
13    yield* triples.transact([
14      { op: "assert", entityId: EntityId.make("customer:acme"), entityType: "Customer", attribute: ":customer/name", value: string("Acme") },
15      { op: "assert", entityId: EntityId.make("engineer:lin"), entityType: "Engineer", attribute: ":engineer/name", value: string("Lin") },
16    ]);
17    const as = (commandId: string) => ({ actor: "agent:triage", commandId, now: 1_800_000_000_000 });
18    yield* runtime.invoke("open-ticket", { ticket: "ticket:1", customer: "customer:acme", title: "VPN drops hourly", priority: 2 }, as("open:1"));
19    yield* runtime.invoke("open-ticket", { ticket: "ticket:2", customer: "customer:acme", title: "Invoice PDF blank", priority: 1 }, as("open:2"));
20    yield* runtime.invoke("open-ticket", { ticket: "ticket:3", customer: "customer:acme", title: "Password reset", priority: 3 }, as("open:3"));
21    yield* runtime.invoke("escalate-ticket", { ticket: "ticket:1", engineer: "engineer:lin", reason: "Needs network access" }, as("escalate:1"));
22    yield* runtime.invoke("close-ticket", { ticket: "ticket:3" }, as("close:3"));
23    return {
24      "open-work": (yield* runtime.query("open-work")).results,
25      escalations: (yield* runtime.query("escalations")).results,
26      "tickets-by-status": (yield* runtime.query("tickets-by-status")).results,
27    };
28  }).pipe(Effect.provide(KvTriples.layer)),
29);
30console.log(JSON.stringify(results, null, 2));
