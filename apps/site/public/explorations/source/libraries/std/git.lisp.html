1; /std/git — provider-neutral Git graph and repository connections.
2; Commit ids identify objects by object format + full oid, across repositories.
3; Branch ids are repository-scoped; names such as main are not global ids.
4; TODO: enforce reference existence, branch/repository consistency and acyclic ancestry.
5; These actions record local facts; they do not execute Git commands.
6 
7(define-entity GitRepository
8  (:doc "A logical Git repository, independent of its hosting provider.")
9  (:field [git-repository/name String {:required true}])
10  (:field [git-repository/default-branch GitBranch]))
11 
12(define-entity GitCommit
13  (:doc "A Git commit object, shared by repositories that contain the same object.")
14  (:field [git-commit/oid String {:required true}])
15  ; Conventional values: sha1, sha256. Strings are not enums in v0.1.
16  (:field [git-commit/object-format String {:required true}])
17  (:field [git-commit/message String {:required true}]))
18 
19(define-entity GitBranch
20  (:doc "A repository-scoped branch with a replaceable tip commit.")
21  (:field [git-branch/repository GitRepository {:required true}])
22  (:field [git-branch/name String {:required true}])
23  (:field [git-branch/tip GitCommit {:required true}]))
24 
25; A merge commit can have multiple parents; position records their order.
26(define-relation git-parent GitCommit GitCommit
27  (:field [git-parent/position Int {:required true}]))
28 
29; Direction: fork -> upstream, dependent -> dependency.
30; Dependencies are modeled project facts, not intrinsic Git object data.
31(define-relation git-fork-of GitRepository GitRepository)
32(define-relation git-depends-on GitRepository GitRepository)
33 
34(define-action git-create-repository
35  (:input [repository String {:required true}])
36  (:input [name String {:required true}])
37  (:returns String)
38  (:do (create GitRepository repository {:git-repository/name name})))
39 
40(define-action git-record-commit
41  (:input [commit String {:required true}])
42  (:input [oid String {:required true}])
43  (:input [objectFormat String {:required true}])
44  (:input [message String {:required true}])
45  (:returns String)
46  (:do (create GitCommit commit
47    {:git-commit/oid oid :git-commit/object-format objectFormat :git-commit/message message})))
48 
49(define-action git-create-branch
50  (:input [branch String {:required true}])
51  (:input [repository String {:required true}])
52  (:input [name String {:required true}])
53  (:input [tip String {:required true}])
54  (:returns String)
55  (:do (create GitBranch branch
56    {:git-branch/repository repository :git-branch/name name :git-branch/tip tip})))
57 
58(define-action git-move-branch
59  (:input [branch String {:required true}])
60  (:input [tip String {:required true}])
61  (:returns String)
62  (:do (changes
63    (clear branch :git-branch/tip)
64    (set branch :git-branch/tip tip))))
65 
66(define-action git-set-default-branch
67  (:input [repository String {:required true}])
68  (:input [branch String {:required true}])
69  (:returns String)
70  (:do (changes
71    (clear repository :git-repository/default-branch)
72    (set repository :git-repository/default-branch branch))))
73 
74(define-action git-record-parent
75  (:input [commit String {:required true}])
76  (:input [parent String {:required true}])
77  (:input [position Int {:required true}])
78  (:returns String)
79  (:do (link git-parent commit parent {:git-parent/position position})))
80 
81(define-action git-record-fork
82  (:input [fork String {:required true}])
83  (:input [upstream String {:required true}])
84  (:returns String)
85  (:do (link git-fork-of fork upstream {})))
86 
87(define-action git-record-dependency
88  (:input [repository String {:required true}])
89  (:input [dependency String {:required true}])
90  (:returns String)
91  (:do (link git-depends-on repository dependency {})))

93(define-datalog-query git-branch-tips
94  (:query
95    {:find ["?repository" "?branch" "?oid"]
96     :where [["?b" ":git-branch/repository" "?r"]
97             ["?r" ":git-repository/name" "?repository"]
98             ["?b" ":git-branch/name" "?branch"]
99             ["?b" ":git-branch/tip" "?c"]
100             ["?c" ":git-commit/oid" "?oid"]]}))
101 
102(define-datalog-query git-repository-dependencies
103  (:query
104    {:find ["?repository" "?dependency"]
105     :where [["?edge" ":git-depends-on/from" "?r"]
106             ["?edge" ":git-depends-on/to" "?d"]
107             ["?r" ":git-repository/name" "?repository"]
108             ["?d" ":git-repository/name" "?dependency"]]}))
