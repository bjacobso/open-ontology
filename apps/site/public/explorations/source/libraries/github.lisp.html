1; /github — GitHub hosting and collaboration layered on /std/git.
2; Local metadata only: no API calls, synchronization or permission enforcement.
3; TODO: validate account kinds, states, repo-local numbers and PR base consistency.
4; TODO: model teams, reviews, checks, releases and link removal when needed.
5(import "/std/git")
6 
7(define-entity GitHubAccount
8  (:doc "A GitHub user or organization, identified independently of its mutable login.")
9  (:field [github-account/node-id String {:required true}])
10  (:field [github-account/login String {:required true}])
11  ; Conventional values: user, organization.
12  (:field [github-account/kind String {:required true}]))
13 
14(define-entity GitHubRepository
15  (:doc "A GitHub hosting record for a provider-neutral Git repository.")
16  (:field [github-repository/node-id String {:required true}])
17  (:field [github-repository/repository GitRepository {:required true}])
18  (:field [github-repository/owner GitHubAccount {:required true}])
19  (:field [github-repository/full-name String {:required true}])
20  (:field [github-repository/url String {:required true}]))
21 
22(define-entity GitHubIssue
23  (:doc "An issue number scoped to its GitHub repository.")
24  (:field [github-issue/repository GitHubRepository {:required true}])
25  (:field [github-issue/number Int {:required true}])
26  (:field [github-issue/title String {:required true}])
27  (:field [github-issue/author GitHubAccount {:required true}])
28  ; Conventional values: open, closed.
29  (:field [github-issue/state String {:required true}]))
30 
31(define-entity GitHubPullRequest
32  (:doc "A proposed change from a head branch, possibly in a fork, into a base branch.")
33  (:field [github-pull-request/repository GitHubRepository {:required true}])
34  (:field [github-pull-request/number Int {:required true}])
35  (:field [github-pull-request/title String {:required true}])
36  (:field [github-pull-request/author GitHubAccount {:required true}])
37  (:field [github-pull-request/base GitBranch {:required true}])
38  (:field [github-pull-request/head GitBranch {:required true}])
39  ; Conventional values: open, closed; merged is a separate fact.
40  (:field [github-pull-request/state String {:required true}])
41  (:field [github-pull-request/merged Bool {:required true}]))

43; Member -> organization. Role/permission fields are descriptive metadata.
44(define-relation github-member-of GitHubAccount GitHubAccount
45  (:field [github-member-of/role String {:required true}]))
46(define-relation github-collaborator GitHubRepository GitHubAccount
47  (:field [github-collaborator/permission String {:required true}]))
48; A reference does not imply that the PR closes the issue.
49(define-relation github-references-issue GitHubPullRequest GitHubIssue)

51(define-action github-record-account
52  (:input [account String {:required true}])
53  (:input [nodeId String {:required true}])
54  (:input [login String {:required true}])
55  (:input [kind String {:required true}])
56  (:returns String)
57  (:do (create GitHubAccount account
58    {:github-account/node-id nodeId :github-account/login login :github-account/kind kind})))

60(define-action github-record-repository
61  (:input [hosting String {:required true}])
62  (:input [nodeId String {:required true}])
63  (:input [repository String {:required true}])
64  (:input [owner String {:required true}])
65  (:input [fullName String {:required true}])
66  (:input [url String {:required true}])
67  (:returns String)
68  (:do (create GitHubRepository hosting
69    {:github-repository/node-id nodeId :github-repository/repository repository
70     :github-repository/owner owner :github-repository/full-name fullName :github-repository/url url})))

72(define-action github-open-issue
73  (:input [issue String {:required true}])
74  (:input [repository String {:required true}])
75  (:input [number Int {:required true}])
76  (:input [title String {:required true}])
77  (:input [author String {:required true}])
78  (:returns String)
79  (:do (create GitHubIssue issue
80    {:github-issue/repository repository :github-issue/number number
81     :github-issue/title title :github-issue/author author :github-issue/state "open"})))

83(define-action github-open-pull-request
84  (:input [pullRequest String {:required true}])
85  (:input [repository String {:required true}])
86  (:input [number Int {:required true}])
87  (:input [title String {:required true}])
88  (:input [author String {:required true}])
89  (:input [base String {:required true}])
90  (:input [head String {:required true}])
91  (:returns String)
92  (:do (create GitHubPullRequest pullRequest
93    {:github-pull-request/repository repository :github-pull-request/number number
94     :github-pull-request/title title :github-pull-request/author author
95     :github-pull-request/base base :github-pull-request/head head
96     :github-pull-request/state "open" :github-pull-request/merged false})))

118(define-datalog-query github-repositories
119  (:query
120    {:find ["?repository" "?owner" "?url"]
121     :where [["?r" ":github-repository/full-name" "?repository"]
122             ["?r" ":github-repository/url" "?url"]
123             ["?r" ":github-repository/owner" "?a"]
124             ["?a" ":github-account/login" "?owner"]]}))

126(define-datalog-query github-forks
127  (:query
128    {:find ["?fork" "?upstream"]
129     :where [["?edge" ":git-fork-of/from" "?f"]
130             ["?edge" ":git-fork-of/to" "?u"]
131             ["?fh" ":github-repository/repository" "?f"]
132             ["?fh" ":github-repository/full-name" "?fork"]
133             ["?uh" ":github-repository/repository" "?u"]
134             ["?uh" ":github-repository/full-name" "?upstream"]]}))

136(define-datalog-query github-pull-request-issues
